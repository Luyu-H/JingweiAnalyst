# ---------------------------------------------------------------
# 1. CMIP6 historical SST: native-grid 1971-2000 climatology
# ---------------------------------------------------------------
models = ["CESM2", "GFDL-ESM4", "MPI-ESM1-2-LR"]
time_range = ["1971-01", "2000-12"]

model_clim = {}
native_mean = {}

for m in models:
    tos = load_regular_data(dataset="cmip6_historical", variable="tos",
                            model=m, time_range=time_range)
    tos = transform_unit(tos, from_unit=tos.attrs.get("units", "K"), to_unit="degC")
    clim = temporal_aggregate(tos, freq="mean")          # day-weighted long-term mean
    model_clim[m] = clim
    # native-ocean global mean, weighted by the model's own areacello
    native_mean[m] = float(
        compute_basic_stat(clim, op="area_weighted_mean", dim=["lat", "lon"]).compute()
    )

# ---------------------------------------------------------------
# 2. WOA23 0 m annual SST climatology, 1971-2000 normal
# ---------------------------------------------------------------
woa = load_climatology_data(dataset="woa23", variable="temperature",
                            period="1971-2000", time_slice="annual",
                            depth_range=[0, 10])
woa_sst = woa.isel(depth=0, drop=True) if "depth" in woa.dims else woa
woa_sst = woa_sst.assign_attrs(units="degC")

woa_native_mean = float(
    compute_basic_stat(woa_sst, op="area_weighted_mean", dim=["lat", "lon"]).compute()
)

# WOA23 1-degree land-sea mask: ocean is (value > 1)
lsm = load_landsea_mask()
ocean = lsm > 1

# ---------------------------------------------------------------
# 3. Regrid each model climatology conservatively to the WOA23 grid
# ---------------------------------------------------------------
regridded = {}
for m in models:
    regridded[m] = regrid_horizontal(model_clim[m], target="woa23", method="conservative")

# ---------------------------------------------------------------
# 4. Common comparison mask (WOA ocean cells valid for every field)
# ---------------------------------------------------------------
mask = build_comparison_mask([regridded[m] for m in models] + [woa_sst],
                             ocean_only=True, extra_mask=ocean)

# ---------------------------------------------------------------
# 5. Comparison statistics on the common cells
# ---------------------------------------------------------------
comp_mean = {}
delta = {}
rmse = {}
corr = {}
bias_field = {}
woa_comp_mean = None

for m in models:
    f = regridded[m].where(mask)
    w = woa_sst.where(mask)

    dm = compute_difference_metrics(data=f, reference=w, dim=["lat", "lon"])
    comp_mean[m] = float(dm["mean_data"])
    delta[m] = float(dm["bias"])
    rmse[m] = float(dm["rmse"])
    bias_field[m] = dm["difference"].assign_attrs(
        units="degC", long_name=f"{m} SST bias (model - WOA23)"
    )
    if woa_comp_mean is None:
        woa_comp_mean = float(dm["mean_reference"])

    c = analyze_correlation(var1=f, var2=w, dim=["lat", "lon"],
                            method="pearson", weighting="area")
    corr[m] = float(c["correlation"])

# ---------------------------------------------------------------
# 6. Ranking by absolute mean bias
# ---------------------------------------------------------------
ranking = sorted(models, key=lambda mm: abs(delta[mm]))

print("Native-grid global mean SST (degC):")
for m in models:
    print(f"  {m:15s} {native_mean[m]:.2f}")
print(f"  {'WOA23 (full grid)':15s} {woa_native_mean:.2f}")

print("\nOn the common WOA23 comparison cells:")
print(f"  WOA23 mean SST: {woa_comp_mean:.2f} degC")
for m in models:
    print(f"  {m:15s} mean={comp_mean[m]:.2f}  delta_SST={delta[m]:+.2f}  "
          f"RMSE={rmse[m]:.3f}  r={corr[m]:.3f}")

print("\nRanking (smallest -> largest |delta_SST|):")
for i, m in enumerate(ranking, 1):
    print(f"  {i}. {m}  (|delta| = {abs(delta[m]):.2f} degC)")

summary = pd.DataFrame({
    "native_global_mean_degC": [native_mean[m] for m in models] + [woa_native_mean],
    "comparison_mean_degC":    [comp_mean[m] for m in models] + [woa_comp_mean],
    "delta_SST_degC":          [delta[m] for m in models] + [np.nan],
    "RMSE_degC":               [rmse[m] for m in models] + [np.nan],
    "spatial_r":               [corr[m] for m in models] + [np.nan],
}, index=models + ["WOA23"])

# --- visualization ---
# ---------------------------------------------------------------
# Figure 1: 2x2 bias maps (shared diverging scale) + WOA23 reference
# ---------------------------------------------------------------
lon = woa_sst["lon"].values
lat = woa_sst["lat"].values

pooled = np.concatenate([bias_field[m].values.ravel() for m in models])
vmax = float(np.nanpercentile(np.abs(pooled), 98))

ref_vals = woa_sst.where(mask).values
rvmin = float(np.nanpercentile(ref_vals, 1))
rvmax = float(np.nanpercentile(ref_vals, 99))

fig, axes = plt.subplots(2, 2, figsize=(14, 8.0), constrained_layout=True)
axes = axes.ravel()

for i, m in enumerate(models):
    ax = axes[i]
    pcm = ax.pcolormesh(lon, lat, bias_field[m].values, cmap="RdBu_r",
                       vmin=-vmax, vmax=vmax, shading="auto")
    ax.set_title(f"{m} − WOA23")
    cb = fig.colorbar(pcm, ax=ax, extend="both")
    cb.set_label("SST bias (°C)")

ax = axes[3]

The three CMIP6 historical simulations reproduce the WOA23 1971–2000 SST climatology well in spatial pattern, with small global-mean offsets but non-uniform regional biases. Using the executed workflow, `tos` was loaded for CESM2, GFDL-ESM4 and MPI-ESM1-2-LR over 1971-01 to 2000-12, converted from K to °C, and temporally aggregated with `temporal_aggregate(freq="mean")` to form the 1971–2000 climatology. WOA23 temperature was loaded as the 1971–2000 annual climatology, depth range `[0,10]`, then the first depth level was selected; it was treated as °C. The WOA23 land–sea mask was loaded with `load_landsea_mask` and ocean cells were defined by `landsea_01 > 1`. Model climatologies were conservatively regridded to the WOA23 1° grid with `regrid_horizontal(target="woa23", method="conservative")`. A common comparison mask was built with `build_comparison_mask(..., ocean_only=True, extra_mask=ocean)`, restricted to cells valid for all three regridded models and WOA23; 40,830 common cells were used. Means, biases and RMSE were computed with `compute_difference_metrics`, and spatial Pearson correlations with `analyze_correlation(..., method="pearson", weighting="area")`. Global/native means were computed with `compute_basic_stat(..., op="area_weighted_mean", dim=["lat","lon"])`. The executed code does not show an explicit `load_areas`/`areacello` argument; the weighted values reported below are those returned by those `compute_basic_stat` calls.

GFDL-ESM4 has the smallest global-mean bias and the lowest RMSE. CESM2 has a moderate warm mean bias, while MPI-ESM1-2-LR has the largest cold mean bias and largest RMSE. All models nevertheless capture the large-scale SST pattern very strongly, with correlations of 0.993–0.997. The bias fields are not uniform: the unweighted spread of the model-minus-WOA23 fields is about 1.03°C standard deviation for CESM2 (range roughly −4.80 to +8.84°C), 0.82°C for GFDL-ESM4 (−4.49 to +7.38°C), and 1.20°C for MPI-ESM1-2-LR (−7.33 to +7.61°C). Thus regional warm and cold biases are larger than the global-mean offsets and partly compensate in the global mean; the spatial maps are important for interpreting model performance beyond the global scalar metrics.
