def _as_float(x):
    if isinstance(x, xr.DataArray):
        return float(x.compute().values)
    return float(x)

def _as_time(x):
    """Coerce a scalar / 0-d DataArray / coordinate mapping to a pandas Timestamp.

    `compute_difference_metrics` returns the location of the largest departure as a
    coordinate mapping (e.g. {'time': '1992-01'}) when only 'time' is reduced, so a
    dict must be unwrapped before it can be turned into a Timestamp.
    """
    if isinstance(x, xr.DataArray):
        x = x.values
    if isinstance(x, dict):
        for key in ("time", "month", "date"):
            if key in x:
                x = x[key]
                break
        else:
            x = next(iter(x.values()))
    arr = np.asarray(x, dtype=object).reshape(-1)
    value = arr[0]
    while isinstance(value, dict):
        value = next(iter(value.values()))
    return pd.Timestamp(value)
