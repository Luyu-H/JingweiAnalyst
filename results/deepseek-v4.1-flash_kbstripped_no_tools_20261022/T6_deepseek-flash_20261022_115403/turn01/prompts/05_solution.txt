def pick_var(ds, var):
    if var in ds.variables:
        return ds[var]
    for name in ds.data_vars:
        if name.lower() == str(var).lower():
            return ds[name]
    for name in ds.data_vars:
        if "bnd" not in name.lower() and ds[name].ndim >= 2:
            return ds[name]
    raise KeyError(f"Could not find variable {var!r} in {list(ds.data_vars)}")


def find_cmip6_files(model, var):
    d = os.path.join(cmip6_root, model, var)
    files = sorted(glob.glob(os.path.join(d, f"{var}_*.nc")))
    if not files:
        files = sorted(glob.glob(os.path.join(d, "*.nc")))
    files = [

def open_cmip6_var(model, var):
    files = find_cmip6_files(model, var)
    das = []
    for f in files:
        ds = xr.open_dataset(f, use_cftime=True, decode_times=True)
        das.append(pick_var(ds, var))
    if len(das) == 1:
        return das[0]
    if all("time" in da.dims for da in das):
        da = xr.concat(das, dim="time")
        da = da.sortby("time")
        return da
    return das[0]


def get_cmip6_climatology(model, var, y0=1971, y1=2000):
    da = open_cmip6_var(model, var)
    if "time" not in da.dims:
        return da.squeeze(drop=True)
    years = da.time.dt.year
    tmask = (years >= y0) & (years <= y1)
    da = da.isel(time=np.where(tmask.values)[0])
    w = da.time.dt.days_in_month
    w_b = w.broadcast_like(da)
    valid = da.notnull()
    num = (da.fillna(0.0) * w_b).sum("time")
    den = (valid.astype(float) * w_b).sum("time")
    clim = (num / den).where(den > 0)
    return clim.compute()


def read_msk_first3(path):
    rows = []
    with open(path, "r") as f:
        for line in f:
            s = line.strip()
            if (not s) or s.startswith("#"):
                continue
            parts = s.replace(",", " ").split()
            vals = []
            for p in parts[:3]:
                try:
                    vals.append(float(p))
                except ValueError:
                    break
            if len(vals) == 3:
                rows.append(vals)
    return np.asarray(rows, dtype=float)

def compute_cell_areas(lat, lon, R=6371000.0):
    lat = np.asarray(lat, dtype=float)
    lon = np.asarray(lon, dtype=float)
    lat_edges = np.empty(len(lat) + 1, dtype=float)
    lat_edges[1:-1] = 0.5 * (lat[:-1] + lat[1:])
    lat_edges[0] = lat[0] - 0.5 * (lat[1] - lat[0])
    lat_edges[-1] = lat[-1] + 0.5 * (lat[-1] - lat[-2])
    lon_edges = np.empty(len(lon) + 1, dtype=float)
    lon_edges[1:-1] = 0.5 * (lon[:-1] + lon[1:])
    lon_edges[0] = lon[0] - 0.5 * (lon[1] - lon[0])
    lon_edges[-1] = lon[-1] + 0.5 * (lon[-1] - lon[-2])
    dlon = np.deg2rad(np.diff(lon_edges))
    lat_rad = np.deg2rad(lat_edges)
    dsin = np.sin(lat_rad[1:]) - np.sin(lat_rad[:-1])
    return (R ** 2) * dsin[:, None] * dlon[None, :]

def nearest_indices_flat(lat_vals, lon_vals, lat_target, lon_target, chunk=20000):
    shape = np.asarray(lat_vals).shape
    lat_flat = np.asarray(lat_vals, dtype=float).ravel()
    lon_flat = np.asarray(lon_vals, dtype=float).ravel()
    n = lat_flat.size
    lat_idx = np.empty(n, dtype=int)
    lon_idx = np.empty(n, dtype=int)
    for start in range(0, n, chunk):
        end = min(start + chunk, n)
        lat_idx[start:end] = np.abs(
            lat_flat[start:end, None] - lat_target[None, :]
        ).argmin(axis=1)
        lon_idx[start:end] = np.abs(
            lon_flat[start:end, None] - lon_target[None, :]
        ).argmin(axis=1)
    return lat_idx.reshape(shape), lon_idx.reshape(shape)


def nitracline_from_field(field, depth_values, depth_dim, top=1000.0):
    z = np.abs(np.asarray(depth_values, dtype=float))
    order = np.argsort(z)
    if not np.array_equal(order, np.arange(len(z))):
        field = field.isel({depth_dim: order})
        z = z[order]

    diff_field = field.diff(depth_dim).rename({depth_dim: "lev_mid"})
    dz = np.diff(z)
    grad = diff_field / xr.DataArray(dz, dims=["lev_mid"])
    mid = xr.DataArray(0.5 * (z[:-1] + z[1:]), dims=["lev_mid"])
    shallow = xr.DataArray(z[:-1], dims=["lev_mid"])
    grad = grad.assign_coords(lev_mid=mid)
    grad = grad.where(shallow <= top)

    idx = grad.fillna(-np.inf).argmax("lev_mid")
    nc = mid.isel(lev_mid=idx)
    nc = nc.where(grad.notnull().any("lev_mid"))
    return nc, grad


def area_mean(da, area, mask=None):
    if mask is not None:
        da = da.where(mask)
        area = area.where(mask)
    valid = da.notnull() & area.notnull() & (area > 0)
    num = (da * area).where(valid).sum()
    den = area.where(valid).sum()
    return float((num / den).compute())


def fill_nearest_2d(field, allowed_mask, max_iter=80):
    filled = field.copy()
    filled[~allowed_mask] = np.nan
    for _ in range(max_iter):
        nan = np.isnan(filled) & allowed_mask
        if not np.any(nan):
            break
        sum_neigh = np.zeros_like(filled)
        count = np.zeros_like(filled)

        neigh = np.full_like(filled, np.nan)
        neigh[1:, :] = filled[:-1, :]
        valid = np.isfinite(neigh)
        sum_neigh[valid] += neigh[valid]
        count[valid] += 1

        neigh = np.full_like(filled, np.nan)
        neigh[:-1, :] = filled[1:, :]
        valid = np.isfinite(neigh)
        sum_neigh[valid] += neigh[valid]
        count[valid] += 1

        neigh = np.roll(filled, 1, axis=1)
        valid = np.isfinite(neigh)
        sum_neigh[valid] += neigh[valid]
        count[valid] += 1

        neigh = np.roll(filled, -1, axis=1)
        valid = np.isfinite(neigh)
        sum_neigh[valid] += neigh[valid]
        count[valid] += 1

        take = nan & (count > 0)
        filled[take] = sum_neigh[take] / count[take]
    return filled


def regrid_conservative(field, area, lat_idx, lon_idx, allowed_mask):
    vals = np.asarray(field.values, dtype=float)
    areas = np.asarray(area.values, dtype=float)
    valid = np.isfinite(vals) & np.isfinite(areas) & (areas > 0)
    nlat, nlon = allowed_mask.shape
    ids = lat_idx * nlon + lon_idx
    num = np.bincount(
        ids[valid],
        weights=(vals[valid] * areas[valid]),
        minlength=nlat * nlon,
    )
    den = np.bincount(
        ids[valid],
        weights=areas[valid],
        minlength=nlat * nlon,
    )
    out = np.full(nlat * nlon, np.nan, dtype=float)
    good = den > 0
    out[good] = num[good] / den[good]
    out = out.reshape(nlat, nlon)
    out = fill_nearest_2d(out, allowed_mask)
    out[~allowed_mask] = np.nan
    return out


def weighted_corr_np(a, b, w, mask):
    a = np.asarray(a, dtype=float)[mask]
    b = np.asarray(b, dtype=float)[mask]
    w = np.asarray(w, dtype=float)[mask]
    valid = np.isfinite(a) & np.isfinite(b) & np.isfinite(w)
    a = a[valid]
    b = b[valid]
    w = w[valid]
    w = w / w.sum()
    ma = np.sum(w * a)
    mb = np.sum(w * b)
    cov = np.sum(w * (a - ma) * (b - mb))
    va = np.sum(w * (a - ma) ** 2)
    vb = np.sum(w * (b - mb) ** 2)
    return float(cov / np.sqrt(va * vb))


def weighted_rmse_np(a, b, w, mask):
    a = np.asarray(a, dtype=float)[mask]
    b = np.asarray(b, dtype=float)[mask]
    w = np.asarray(w, dtype=float)[mask]
    valid = np.isfinite(a) & np.isfinite(b) & np.isfinite(w)
    a = a[valid]
    b = b[valid]
    w = w[valid]
    return float(np.sqrt(np.sum(w * (a - b) ** 2) / np.sum(w)))


def simple_density(theta, salt, depth):
    # Lightweight in-situ density estimate in kg m-3, sufficient for the
    # volume-to-mass conversion in Figure 2.  theta in degC, salt ~35.
    rho = 1027.0 - 0.15 * theta + 0.78 * (salt - 35.0) + 0.0045 * depth
    return rho.clip(min=1000.0, max=1050.0)


def section_band(da, area, lat2d, lon2d_180, z, depth_dim,
                 lat_min=-2.0, lat_max=2.0,
                 lon_min=120.0, lon_max=280.0, dlon=1.0):
    ydim, xdim = lat2d.dims
    da_t = da.transpose(depth_dim, ydim, xdim)
    area_t = area.transpose(ydim, xdim)
    latv = lat2d.transpose(ydim, xdim).values
    lonv = ((lon2d_180 + 360.0) % 360.0).transpose(ydim, xdim).values
    areav = area_t.values
    dav = da_t.values

    sel = (
        (latv >= lat_min) & (latv <= lat_max) &
        (lonv >= lon_min) & (lonv <= lon_max) &
        np.isfinite(areav) & (areav > 0)
    )

    lon_edges = np.arange(lon_min - 0.5, lon_max + 0.5 + 1e-9, dlon)
    centers = 0.5 * (lon_edges[:-1] + lon_edges[1:])
    nlonb = len(centers)
    bin_idx = np.floor((lonv - (lon_min - 0.5)) / dlon).astype(int)
    bin_idx = np.clip(bin_idx, 0, nlonb - 1)

    idx_sel = bin_idx[sel]
    area_sel = areav[sel]
    out = np.full((dav.shape[0], nlonb), np.nan, dtype=float)

    for k in range(dav.shape[0]):
        vals = dav[k][sel]
        valid = np.isfinite(vals) & np.isfinite(area_sel)
        if not np.any(valid):
            continue
        idx = idx_sel[valid]
        w = area_sel[valid]
        num = np.bincount(idx, weights=vals[valid] * w, minlength=nlonb)
        den = np.bincount(idx, weights=w, minlength=nlonb)
        good = den > 0
        out[k, good] = num[good] / den[good]

    return {
        "data": out,
        "depth": np.asarray(z, dtype=float),
        "lon": centers,
        "lon_edges": lon_edges,
    }


def nitracline_section(data, depth, top=1000.0):
    z = np.abs(np.asarray(depth, dtype=float))
    order = np.argsort(z)
    if not np.array_equal(order, np.arange(len(z))):
        data = data[order, :]
        z = z[order]
    dz = np.diff(z)
    grad = np.diff(data, axis=0) / dz[:, None]
    mid = 0.5 * (z[:-1] + z[1:])
    allowed = z[:-1] <= top
    grad = np.where(allowed[:, None], grad, np.nan)
    nc = np.full(data.shape[1], np.nan, dtype=float)
    for j in range(data.shape[1]):
        col = grad[:, j]
        if np.any(np.isfinite(col)):
            nc[j] = mid[np.nanargmax(col)]
    return nc

The CMIP6 models generally place the nitracline too shallow compared with the WOA23 all-period annual nitrate climatology. **MPI-ESM1-2-LR is the best of the three models by the stated criterion** (lowest area-weighted RMSE, supported by the highest spatial correlation), and it is the only model that reproduces the WOA23 ordering of nitracline depth across the Pacific, Indian and Atlantic basins. However, all models have modest spatial correlations and large pointwise RMSEs, so the agreement is weak even for MPI-ESM1-2-LR.

## Methodology actually used
- **WOA23 reference:** The code opened `woa23_all_n00_01.nc`, selected the objectively analysed nitrate variable `n_an`, removed the singleton time dimension, kept fill values as missing, and used the annual all-period nitrate field as the observational reference.
- **Model climatology:** Monthly `no3` for CESM2, GFDL-ESM4 and MPI-ESM1-2-LR was selected for 1971–2000 and averaged over time using `days_in_month` weights from each model’s calendar. The nitrate field was time-averaged first; nitraclines were then located from the resulting climatological profiles.
- **Nitracline definition:** For each nitrate profile, the code computed vertical nitrate gradients between adjacent native model or WOA levels, restricted allowed gradients to those whose shallower level was at or above 1000 m, and assigned the nitracline depth to the midpoint of the pair with the maximum downward nitrate gradient. Retaining levels below 1000 m allowed the deepest permitted gradient to use the level just below 1000 m.
- **Grids and means:** Global and basin means were computed on each source’s native horizontal grid with its own area weights: model `areacello` for CMIP6 and WOA cell areas computed from the WOA latitude/longitude centers. Basin labels were taken by parsing the WOA `basinmask_01.msk` and `landsea_01.msk`, mapping marginal seas to parent ocean labels by custom geographic rules, and carrying those labels to each model grid by nearest WOA grid indices. Southern/Arctic parent labels were excluded.
- **Comparison fields:** Model nitracline fields were regridded horizontally to the WOA 1° grid by area-weighted aggregation into nearest WOA target cells, followed by nearest filling within the WOA ocean mask. A shared valid mask was then built from the finite WOA nitracline, WOA ocean mask, and finite regridded nitraclines from all three models; it contained **40,990 cells**. Correlation and RMSE were computed on that shared mask using WOA 1° cell areas as weights. No vertical regridding was applied.
- **Figure 2 preparation:** The code formed 2°S–2°N longitude–depth section arrays for 120°E–80°W, area-weighted into 1° longitude bins. Model nitrate was converted from mol m⁻³ to µmol kg⁻¹ using a lightweight linear in-situ density estimate based on `thetao`, `so` and depth, not a TEOS-10 potential-density calculation. Nitracline sections were computed from the native-unit profiles because nitracline depth is scale-invariant.
- **No plotting was executed.** The code imported Matplotlib and prepared fields for Figures 1 and 2, but it did not call plotting routines. Therefore no visualization was generated.

## Numerical results

### Global mean nitracline depth
| Source | Mean depth (m) | Difference from WOA23 (m) |
|---|---:|---:|
| WOA23 | 164 | — |
| CESM2 | 146 | -17 |
| GFDL-ESM4 | 133 | -31 |
| MPI-ESM1-2-LR | 136 | -28 |

All three models simulate a globally shallower nitracline than WOA23. GFDL-ESM4 is shallowest and has the largest global mean bias; MPI-ESM1-2-LR is closest to the observed global mean.

### Basin mean nitracline depths
| Source | Pacific (m) | Atlantic (m) | Indian (m) |
|---|---:|---:|---:|
| WOA23 | 205 | 160 | 166 |
| CESM2 | 159 | 107 | 163 |
| GFDL-ESM4 | 157 | 90 | 171 |
| MPI-ESM1-2-LR | 188 | 112 | 122 |

Model-minus-WOA23 differences:
| Model | Pacific Δ (m) | Atlantic Δ (m) | Indian Δ (m) |
|---|---:|---:|---:|
| CESM2 | -46 | -53 | -3 |
| GFDL-ESM4 | -49 | -70 | +5 |
| MPI-ESM1-2-LR | -18 | -48 | -44 |

The largest biases are generally in the Atlantic, where all models place the nitracline much too shallow. The Pacific is also too shallow in all models, with MPI-ESM1-2-LR closest. The Indian Ocean bias varies strongly: CESM2 is nearly unbiased, GFDL-ESM4 is slightly too deep, and MPI-ESM1-2-LR is much too shallow.

### Spatial comparison on the WOA 1° grid
| Model | Area-weighted Pearson correlation | Area-weighted RMSE (m) |
|---|---:|---:|
| CESM2 | 0.372 | 165.1 |
| GFDL-ESM4 | 0.459 | 166.2 |
| MPI-ESM1-2-LR | 0.573 | 156.2 |

By lowest area-weighted RMSE, **MPI-ESM1-2-LR is the best model**, and it also has the highest spatial correlation. CESM2 has lower RMSE than GFDL-ESM4 but lower correlation. All correlations are modest and all RMSEs exceed 150 m, indicating substantial pointwise disagreement.

### Basin ordering
WOA23 orders the basins from deepest to shallowest as **Pacific > Indian > Atlantic**.  
Model orders from the reported means are:
- CESM2: Indian > Pacific > Atlantic — does **not** match.
- GFDL-ESM4: Indian > Pacific > Atlantic — does **not** match.
- MPI-ESM1-2-LR: Pacific > Indian > Atlantic — **matches WOA23**.

Thus MPI-ESM1-2-LR is the only model reproducing the observed ordering of nitracline depth across the three basins.

## Important interpretation
The models are broadly biased toward a shallower nitracline than the WOA23 all-period nitrate climatology, especially in the Atlantic and Pacific. MPI-ESM1-2-LR performs best by RMSE and correlation and reproduces the basin-depth ordering, but it still underestimates the global mean depth by about 28 m and strongly underestimates the Indian Ocean nitracline depth. CESM2 and GFDL-ESM4 fail to reproduce the WOA23 basin order by placing the Indian Ocean deeper than the Pacific. The comparison also carries an unavoidable period mismatch: WOA23 is an all-time-period annual composite, whereas the model fields were averaged over 1971–2000.
