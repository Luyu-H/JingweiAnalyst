import numpy as np

import xarray as xr
import matplotlib.pyplot as plt

CMIP_ROOT = "/data1/lorraine-20419/benchmark_cmip6/historical"
ERSST_FILE = "/data1/lorraine-20419/ocean_data/ERSSTv5/sst.mnmean.nc"

MODEL_INFO = {

    "GFDL-ESM4": {"lat_coord": "lat", "lon_coord": "lon"},
    "MPI-ESM1-2-LR": {"lat_coord": "latitude", "lon_coord": "longitude"},
}
MODEL_ORDER = ["CESM2", "GFDL-ESM4", "MPI-ESM1-2-LR"]
SOURCE_ORDER = ["ERSSTv5"] + MODEL_ORDER

MONTH_ABBR = ["Jan", "Feb", "Mar", "Apr", "May", "Jun",

MONTH_NAMES = ["January", "February", "March", "April", "May", "June",
               "July", "August", "September", "October", "November", "December"]

target_time = pd.date_range("1950-01-01", "2014-12-01", freq="MS")
target_months = target_time.month.values
N_MONTHS = len(target_time)
if N_MONTHS != 780:
    raise RuntimeError(f"Expected 780 months, got {N_MONTHS}")


def _variant_from_path(path):
    m = re.search(r"_(r\d+i\d+p\d+f\d+)_", os.path.basename(path))
    return m.group(1) if m else None


def _standardize_time(da):
    da = da.sortby("time")
    tvals = da["time"].values
    if np.issubdtype(tvals.dtype, np.datetime64):
        idx = pd.DatetimeIndex(tvals)
        years = idx.year.values
        months = idx.month.values
    else:
        years = np.array([v.year for v in tvals])
        months = np.array([v.month for v in tvals])

    mask = (
        ((years > 1950) | ((years == 1950) & (months >= 1)))
        & ((years < 2014) | ((years == 2014) & (months <= 12)))
    )
    da = da.isel(time=mask)
    if da.sizes["time"] != N_MONTHS:
        raise ValueError(
            f"Expected {N_MONTHS} months for 1950-01..2014-12, got {da.sizes['time']}"
        )
    da = da.assign_coords(time=target_time)
    return da


def _get_surface(da):
    for d in ("lev", "olevel", "depth", "deptht"):
        if d in da.dims:
            return da.isel({d: 0}, drop=True)
    return da


def _open_fixed(model, var):
    files = sorted(glob.glob(os.path.join(CMIP_ROOT, model, var, "*.nc")))
    if not files:
        return None, None
    ds = xr.open_dataset(files[0], decode_times=False)
    if var in ds:
        name = var
    else:
        candidates = [v for v in ds.data_vars if not v.endswith("_bnds")]
        if not candidates:
            raise KeyError(f"No data variable found in {files[0]}")
        name = candidates[0]
    return ds, ds[name]


def _regional_mean(field, area, lat_coord, lon_coord):
    lat = field[lat_coord]
    lon = field[lon_coord]
    lat2d, lon2d = xr.broadcast(lat, lon)
    lon360 = lon2d % 360.0

    # Nino 3.4: 5S-5N, 170W-120W = 190E-240E on a 0-360 longitude axis.
    box = (
        (lat2d >= -5.0)
        & (lat2d <= 5.0)
        & (lon360 >= 190.0)
        & (lon360 <= 240.0)
    )

    area2d = area.broadcast_like(lat2d)
    area_box = area2d.where(box)

    valid = np.isfinite(field)
    weights = area_box.where(valid)

    spatial_dims = [d for d in field.dims if d != "time"]
    num = (field * weights).sum(dim=spatial_dims, skipna=True)
    den = weights.sum(dim=spatial_dims, skipna=True)

    series = (num / den).compute()
    area_series = den.compute()
    return series.values, area_series.values


def load_cmip6_tos(model):
    info = MODEL_INFO[model]
    files = sorted(glob.glob(os.path.join(CMIP_ROOT, model, "tos", "*.nc")))
    if not files:
        raise FileNotFoundError(f"No tos files for {model}")

    variants = {_variant_from_path(f) for f in files if _variant_from_path(f)}
    if len(variants) != 1:
        raise ValueError(f"Expected one historical variant for {model}, found {variants}")
    variant = next(iter(variants))

    ds = xr.open_mfdataset(
        files, combine="by_coords", use_cftime=True, decode_times=True
    )
    field = _standardize_time(ds["tos"])
    field = _get_surface(field)

    area_ds, area = _open_fixed(model, "areacello")
    if area_ds is None:
        raise FileNotFoundError(f"No areacello for {model}")
    area = area.squeeze(drop=True)
    area_source = f"{model} areacello"

    sft_ds, sft = _open_fixed(model, "sftof")
    if sft_ds is not None:
        sft = sft.squeeze(drop=True)
        area = area.where(sft > 0)
        area_source = f"{model} areacello masked by sftof>0"

    series, area_series = _regional_mean(
        field, area, info["lat_coord"], info["lon_coord"]
    )

    ds.close()
    area_ds.close()
    if sft_ds is not None:
        sft_ds.close()

    return series, area_series, variant, area_source


def load_ersst():
    ds = xr.open_dataset(ERSST_FILE, use_cftime=False, decode_times=True)
    field = _standardize_time(ds["sst"])

    lat = ds["lat"]
    lon = ds["lon"]
    R = 6371000.0
    dlat = np.deg2rad(2.0)
    dlon = np.deg2rad(2.0)
    lat_rad = np.deg2rad(lat)
    area_lat = R**2 * dlon * (
        np.sin(lat_rad + dlat / 2.0) - np.sin(lat_rad - dlat / 2.0)
    )
    area = area_lat * xr.ones_like(lon)
    area_source = "ERSSTv5 regular 2-degree cell areas (R=6371 km)"

    series, area_series = _regional_mean(field, area, "lat", "lon")
    ds.close()
    return series, area_series, area_source


def _compute_stats(series, label):
    series = np.asarray(series, dtype=float)
    if series.size != N_MONTHS:
        raise ValueError(f"Expected {N_MONTHS} values, got {series.size}")

    clim = np.array(
        [np.nanmean(series[target_months == m]) for m in range(1, 13)]
    )
    anomaly = series - clim[target_months - 1]

    overall_std = float(np.nanstd(anomaly, ddof=0))
    seasonal_std = np.array(
        [np.nanstd(anomaly[target_months == m], ddof=0) for m in range(1, 13)]
    )

    peak_idx = int(np.nanargmax(seasonal_std))
    min_idx = int(np.nanargmin(seasonal_std))

    exceed_count = int(np.sum(np.abs(anomaly) > 1.0))
    exceed_pct = float(exceed_count / N_MONTHS * 100.0)

    return {
        "label": label,
        "series": series,
        "anomaly": anomaly,
        "climatology": clim,
        "overall_std": overall_std,
        "seasonal_std": seasonal_std,
        "peak_month": peak_idx + 1,
        "peak_std": float(seasonal_std[peak_idx]),
        "min_month": min_idx + 1,
        "min_std": float(seasonal_std[min_idx]),
        "peak_min_ratio": float(seasonal_std[peak_idx] / seasonal_std[min_idx]),
        "exceed_count": exceed_count,
        "exceed_pct": exceed_pct,
    }


def _add_area_stats(stats, area_series, area_source):
    area_m2 = np.asarray(area_series, dtype=float)
    stats["area_source"] = area_source
    stats["area_mean_1e6_km2"] = float(np.nanmean(area_m2) / 1e12)
    stats["area_min_1e6_km2"] = float(np.nanmin(area_m2) / 1e12)
    stats["area_max_1e6_km2"] = float(np.nanmax(area_m2) / 1e12)

- CMIP6 input: historical `tos` files were opened with `xr.open_mfdataset`, and the single historical variant present in the paths was used. The code found one variant for each model: **r1i1p1f1** for CESM2, GFDL-ESM4, and MPI-ESM1-2-LR. If a vertical dimension was present, the first level was selected.
- Area weighting: each source was averaged with its own cell areas. For CESM2 and MPI-ESM1-2-LR the `areacello` field was additionally masked by `sftof > 0`; for GFDL-ESM4 the area source was `areacello` directly. For ERSSTv5, regular 2-degree spherical cell areas were computed from latitude spacing and longitude spacing with **R = 6371 km**.
- Region: latitude from **−5° to +5°**, longitude **190°E to 240°E** on a 0–360 longitude axis, equivalent to **170°W–120°W**. Weighted means were computed as \(\sum(\mathrm{SST}\times\mathrm{area}) / \sum(\mathrm{area})\), with weights NaN-restricted to valid SST cells in each month.
- Baseline: each source’s anomaly was defined from **its own 1950–2014 calendar-month climatology**. The climatology for each calendar month is the plain mean of that month over the 65 years. No observed climatology or WOA climatology was applied to the models.
- Standard deviations: **population standard deviations**, `ddof=0`. Seasonal standard deviations were computed across the 65 years for each calendar month.
- Exceedance: strict threshold **|Nino 3.4 anomaly| > 1.0 °C**, counted as months out of 780, with no event-detection or minimum-duration rule.
- No lagged correlation or event-by-event comparison was performed; all metrics are contemporaneous monthly statistics.

The average Nino 3.4 box area over which each index was computed differs by source:
- ERSSTv5: **6.42 × 10⁶ km²**
- CESM2: **6.35 × 10⁶ km²**
- GFDL-ESM4: **6.19 × 10⁶ km²**
- MPI-ESM1-2-LR: **6.21 × 10⁶ km²**

The area series were constant through time after validity masking in these runs.

| Source | Peak month | Peak std (°C) | Min month | Min std (°C) | Peak/min ratio | Reproduces observed peak month? | Peak-strength ratio vs ERSSTv5 | Peak/min ratio relative to ERSSTv5 |
|---|---|---:|---|---:|---:|---|---:|---:|

| CESM2 | **January** | **1.62** | **June** | **0.94** | **1.72** | **No** | **1.45** | **0.92** |
| GFDL-ESM4 | **December** | **1.25** | **July** | **0.75** | **1.67** | **Yes** | **1.11** | **0.89** |
| MPI-ESM1-2-LR | **December** | **1.03** | **July** | **0.71** | **1.45** | **Yes** | **0.92** | **0.77** |

Interpretation:
- **CESM2** does not reproduce the observed December peak; its strongest seasonal amplitude occurs in January. It has a much stronger absolute December/January seasonal amplitude than observed, but its peak-to-minimum seasonal contrast is slightly weaker than observed.
- **GFDL-ESM4** reproduces the December peak month. Its December amplitude is about 11% stronger than observed, while its peak/min seasonal contrast is weaker than observed.
- **MPI-ESM1-2-LR** reproduces the December peak month, but its December amplitude is slightly weaker than observed and its seasonal peak/min contrast is substantially weaker than observed.

Overall, MPI-ESM1-2-LR matches the overall ENSO amplitude most closely, GFDL-ESM4 is intermediate, and CESM2 simulates an ENSO that is too energetic. All three models show weaker seasonal phase locking, measured as the seasonal peak/min ratio, than ERSSTv5.

## Important insights and caveats

The comparison is statistical rather than event-by-event. The CMIP6 historical runs are coupled simulations and were not expected to reproduce the observed year-to-year timing of individual ENSO events. Only one historical member per model was available, so internal variability cannot be separated from forced behaviour with these data. In addition, each source is on its own native grid with its own area field, and the Nino 3.4 box area differs by source; this affects small amplitude differences and means that fine-scale features are not strictly comparable across resolutions. The reported standard deviations, biases, ratios, and exceedance percentages are based on unrounded anomaly series, with rounding applied only for reporting.
