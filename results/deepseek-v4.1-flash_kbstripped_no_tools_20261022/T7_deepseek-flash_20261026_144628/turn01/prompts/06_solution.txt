def find_coord(ds, candidates):
    for c in candidates:
        if c in ds.coords or c in ds.variables:
            return c
    raise KeyError(f"None of {candidates} found in dataset")


def to_m(values, units):
    """Convert a depth array to metres according to its declared units."""
    u = str(units).lower().replace(" ", "")
    if "centimeter" in u or "centimetre" in u or u == "cm" or u.endswith("cm"):
        return values * 0.01
    if "millimeter" in u or "millimetre" in u or u == "mm" or u.endswith("mm"):
        return values * 0.001
    if "kilometer" in u or "kilometre" in u or u == "km" or u.endswith("km"):
        return values * 1000.0
    return values


def get_depth_info(ds, lev_name):
    """Return level centres and layer bounds in metres."""
    lev = ds[lev_name]
    lev_m = to_m(np.asarray(lev.values, dtype=float), lev.attrs.get("units", ""))

    bvals = None
    bname = lev_name + "_bnds"
    if bname in ds:
        b = ds[bname]
        if "time" in b.dims:
            b = b.isel(time=0)
        bvals = to_m(np.asarray(b.values, dtype=float), b.attrs.get("units", ""))

    if bvals is None:
        n = len(lev_m)
        edges = np.empty((n, 2), dtype=float)
        if n > 1:
            mids = 0.5 * (lev_m[:-1] + lev_m[1:])
            edges[1:, 0] = mids
            edges[:-1, 1] = mids
            edges[0, 0] = lev_m[0] - 0.5 * (lev_m[1] - lev_m[0])
            edges[-1, 1] = lev_m[-1] + 0.5 * (lev_m[-1] - lev_m[-2])
        else:
            edges[0] = [lev_m[0] - 1.0, lev_m[0] + 1.0]
        bvals = edges

    if bvals.ndim != 2 or bvals.shape[0] != len(lev_m):
        if bvals.ndim == 2 and bvals.shape[1] == len(lev_m):
            bvals = bvals.T
        else:
            raise ValueError("Depth bounds have incompatible shape")

    lo = np.minimum(bvals[:, 0], bvals[:, 1])
    hi = np.maximum(bvals[:, 0], bvals[:, 1])
    return lev_m, lo, hi


def umol_kg_factor(units, rho=RHO):
    """Factor converting the declared concentration units to umol kg-1."""
    u = str(units).lower().replace(" ", "")
    if "umol" in u or "µmol" in u:
        if "kg" in u:
            return 1.0
        if "m-3" in u or "m^-3" in u or "m**-3" in u:
            return 1.0 / rho
    if "mmol" in u:
        return 1000.0 / rho
    if "mol" in u:
        return 1.0e6 / rho
    return 1.0e6 / rho

def classify_basin_centroid(lon, lat):
    """Fold marginal seas into their parent ocean using a code centroid."""
    # Excluded basins
    if lat < -60.0:
        return "Southern"
    if lat > 66.0:
        return "Arctic"

    # Atlantic marginal seas: Mediterranean, Black Sea, Baltic, North Sea
    if (-10.0 <= lon <= 50.0) and (30.0 <= lat <= 72.0):
        return "Atlantic"
    # Hudson Bay
    if (-105.0 <= lon <= -65.0) and (45.0 <= lat <= 70.0):
        return "Atlantic"

    # Indian marginal seas: Red Sea, Persian Gulf
    if (30.0 <= lon <= 65.0) and (10.0 <= lat <= 35.0):
        return "Indian"

    # Pacific marginal seas: South China Sea, Japan Sea, Okhotsk, Bering
    if (100.0 <= lon <= 180.0) and (0.0 <= lat <= 65.0):
        return "Pacific"
    if (-180.0 <= lon <= -150.0) and (45.0 <= lat <= 65.0):
        return "Pacific"

    # Major basins by longitude
    if -70.0 <= lon <= 20.0:
        return "Atlantic"
    if 20.0 < lon <= 100.0:
        return "Indian"
    return "Pacific"


def read_basin_mask(path):
    """Read WOA basin mask, robust to space-separated, fixed-width,
    Fortran D exponents, and missing-value markers such as '*' or 'NA'."""
    rows = []
    with open(path, "r") as f:
        for line in f:
            s = line.strip()
            if not s or s.startswith("#"):
                continue
            if s[0].isalpha():
                continue
            # remove inline comments
            s = s.split("#")[0].split("!")[0]
            # Fortran exponent
            s = s.replace("D", "E").replace("d", "e")
            # asterisks -> NaN marker
            s = re.sub(r"\*+", " nan ", s)
            # common missing markers
            s = re.sub(r"\bNA\b", " nan ", s, flags=re.IGNORECASE)
            # extract all numbers and nan tokens
            tokens = re.findall(
                r"[-+]?(?:\d+\.?\d*|\.\d+)(?:[Ee][-+]?\d+)?|\bnan\b",
                s,
                flags=re.IGNORECASE,
            )
            vals = []
            for t in tokens:
                t_lower = t.lower()
                if t_lower == "nan":
                    vals.append(np.nan)
                else:
                    try:
                        vals.append(float(t))
                    except ValueError:
                        pass
            if len(vals) >= 3:
                rows.append(vals)
    if not rows:
        raise ValueError("No numeric rows in basin mask")
    maxlen = max(len(r) for r in rows)
    arr = np.full((len(rows), maxlen), np.nan, dtype=float)
    for i, r in enumerate(rows):
        arr[i, : len(r)] = r
    return arr


def detect_latlon_cols(arr):
    ncol = arr.shape[1]
    candidates = list(range(min(4, ncol)))
    for i in candidates:
        if not np.any(np.isfinite(arr[:, i])):
            continue
        if np.nanmin(arr[:, i]) >= -90.1 and np.nanmax(arr[:, i]) <= 90.1:
            for j in candidates:
                if j == i:
                    continue
                if not np.any(np.isfinite(arr[:, j])):
                    continue
                if np.nanmin(arr[:, j]) >= -180.1 and np.nanmax(arr[:, j]) <= 360.1:
                    if np.nanmax(np.abs(arr[:, i] - arr[:, j])) > 1.0:
                        return i, j
    return 0, 1


def get_value_cols(arr, lat_col, lon_col):
    cols = [c for c in range(arr.shape[1]) if c not in (lat_col, lon_col)]
    if lat_col > 0 and 0 in cols:
        col0 = arr[:, 0]
        if np.all(np.isfinite(col0)) and np.allclose(col0, np.round(col0)):
            if np.all(np.diff(col0) >= 0):
                cols.remove(0)
    return cols


def sample_basin_grid(lat2d, lon2d, basin_grid, lat_centers, lon_centers):
    lat = np.asarray(lat2d)
    lon = np.asarray(lon2d)
    lon = ((lon + 180.0) % 360.0) - 180.0
    dlat = np.median(np.diff(lat_centers))
    dlon = np.median(np.diff(lon_centers))
    i = np.rint((lat - lat_centers[0]) / dlat).astype(int)
    j = np.rint((lon - lon_centers[0]) / dlon).astype(int)
    i = np.clip(i, 0, basin_grid.shape[0] - 1)
    j = np.clip(j, 0, basin_grid.shape[1] - 1)
    return basin_grid[i, j]


def vol_exceedance(np_ratio, vol, cond, extra_mask=None):
    m = np.isfinite(np_ratio) & np.isfinite(vol) & (vol > 0.0)
    if extra_mask is not None:
        m = m & extra_mask
    total = np.nansum(vol[m])
    num = np.nansum(vol[m & cond])
    pct = 100.0 * num / total if total > 0.0 else np.nan
    return pct, total, num


def compute_stats(np_ratio, nitrate, phosphate, vol, basin=None):
    valid = (
        np.isfinite(np_ratio)
        & np.isfinite(nitrate)
        & np.isfinite(phosphate)
        & np.isfinite(vol)
        & (vol > 0.0)
    )
    total_vol = np.nansum(vol[valid])
    if total_vol <= 0.0:
        return {
            "mean_np": np.nan,
            "ratio_means": np.nan,
            "mean_nitrate": np.nan,
            "mean_phosphate": np.nan,
            "ols_slope": np.nan,
            "ols_intercept": np.nan,
            "r2": np.nan,
            "pct_14_18": np.nan,
            "pct_lt12": np.nan,
            "basin_means": {"Atlantic": np.nan, "Pacific": np.nan, "Indian": np.nan},
            "basin_exceedance": {},
            "valid_count": 0,
            "total_volume": 0.0,
        }

    mean_np = np.nansum(np_ratio[valid] * vol[valid]) / total_vol
    mean_n = np.nansum(nitrate[valid] * vol[valid]) / total_vol
    mean_p = np.nansum(phosphate[valid] * vol[valid]) / total_vol
    ratio_means = mean_n / mean_p if mean_p != 0.0 else np.nan

    x = phosphate[valid]
    y = nitrate[valid]
    a, b = np.polyfit(x, y, 1)
    ypred = a * x + b
    ss_res = np.sum((y - ypred) ** 2)
    ss_tot = np.sum((y - np.mean(y)) ** 2)
    r2 = 1.0 - ss_res / ss_tot if ss_tot > 0.0 else np.nan

    pct_14_18, vol_14_18, num_14_18 = vol_exceedance(
        np_ratio, vol, (np_ratio > 14.0) & (np_ratio < 18.0)
    )
    pct_lt12, vol_lt12, num_lt12 = vol_exceedance(np_ratio, vol, np_ratio < 12.0)

    basin_means = {}
    for bname, bval in [("Atlantic", 1), ("Pacific", 2), ("Indian", 3)]:
        if basin is not None:
            bm = basin == bval
        else:
            bm = np.ones_like(vol, dtype=bool)
        m = valid & bm
        if np.any(m):
            basin_means[bname] = np.nansum(np_ratio[m] * vol[m]) / np.nansum(vol[m])
        else:
            basin_means[bname] = np.nan

    basin_exceedance = {}
    for bname, bval in [("Pacific", 2), ("Indian", 3)]:
        if basin is not None:
            bm = basin == bval
        else:
            bm = np.ones_like(vol, dtype=bool)
        pct_lt, vtot_lt, vnum_lt = vol_exceedance(
            np_ratio, vol, np_ratio < 12.0, extra_mask=bm
        )
        pct_mid, vtot_mid, vnum_mid = vol_exceedance(
            np_ratio, vol, (np_ratio > 14.0) & (np_ratio < 18.0), extra_mask=bm
        )
        basin_exceedance[bname] = {
            "lt12_pct": pct_lt,
            "lt12_volume": vtot_lt,
            "lt12_volume_below": vnum_lt,
            "14_18_pct": pct_mid,
            "14_18_volume": vtot_mid,
            "14_18_volume_between": vnum_mid,
        }

    return {
        "mean_np": mean_np,
        "ratio_means": ratio_means,
        "mean_nitrate": mean_n,
        "mean_phosphate": mean_p,
        "ols_slope": a,
        "ols_intercept": b,
        "r2": r2,
        "pct_14_18": pct_14_18,
        "vol_14_18": vol_14_18,
        "vol_between_14_18": num_14_18,
        "pct_lt12": pct_lt12,
        "vol_lt12": vol_lt12,
        "vol_below_12": num_lt12,
        "basin_means": basin_means,
        "basin_exceedance": basin_exceedance,
        "valid_count": int(np.sum(valid)),
        "total_volume": total_vol,
    }

def open_woa(var):
    if var == "nitrate":
        subdir = "nitrate/all_time_periods"
        tag = "all"
        prefix = "n"
    elif var == "phosphate":
        subdir = "phosphate/all_time_periods"
        tag = "all"
        prefix = "p"
    else:
        raise ValueError(var)

    path = os.path.join(
        woa_root, subdir, f"woa23_{tag}_{prefix}00_01.nc"
    )
    ds = xr.open_dataset(path, decode_times=False)

    candidates = [f"{prefix}_an", var, prefix, f"{prefix}_mn"]
    vname = None
    for c in candidates:
        if c in ds.data_vars:
            vname = c
            break
    if vname is None:
        vname = list(ds.data_vars)[0]

    da = ds[vname]
    if "time" in da.dims:
        da = da.isel(time=0)
    da = da.where(np.abs(da) < FILL_LIMIT)
    return da, ds, vname

def load_cmip6_clim(model, var):
    files = sorted(glob.glob(os.path.join(cmip6_root, model, var, "*.nc")))
    if len(files) == 0:
        return None, None

    ds = xr.open_mfdataset(
        files,
        combine="by_coords",
        use_cftime=True,
        decode_times=True,
    )

    vname = var if var in ds.data_vars else None
    if vname is None:
        for v in ds.data_vars:
            if v.lower() == var.lower():
                vname = v
                break
    if vname is None:
        vname = list(ds.data_vars)[0]

    da = ds[vname]
    units = da.attrs.get("units", "")

    time = ds["time"]
    mask_t = ((time.dt.year >= 1971) & (time.dt.year <= 2000)).values
    da = da.isel(time=mask_t)
    da = da.where(np.abs(da) < FILL_LIMIT)

    clim = da.groupby("time.month").mean("time")
    annual = clim.mean("month")

    factor = umol_kg_factor(units, RHO)
    annual = annual * factor
    annual.attrs["units"] = "micromoles_per_kilogram"
    return annual, ds


def load_cmip6_fixed(model, var):
    files = sorted(glob.glob(os.path.join(cmip6_root, model, var, "*.nc")))
    if len(files) == 0:
        return None, None

    ds = xr.open_mfdataset(

        use_cftime=True,
        decode_times=True,
    )

    vname = var if var in ds.data_vars else list(ds.data_vars)[0]
    da = ds[vname]

    if "time" in da.dims:
        time = ds["time"]

def regrid_to_regular(lat2d, lon2d, field):
    """Nearest/linear interpolation of a curvilinear field to 1 deg lon-lat."""
    lat = np.asarray(lat2d).ravel()
    lon = np.asarray(lon2d).ravel()
    vals = np.asarray(field).ravel()

    lon = ((lon + 180.0) % 360.0) - 180.0
    valid = np.isfinite(vals) & np.isfinite(lat) & np.isfinite(lon)

    pts = np.column_stack([lon[valid], lat[valid]])
    vals_v = vals[valid]

    # Duplicate points across the dateline for interpolation near +/-180
    pts_all = np.vstack(
        [pts, pts + np.array([360.0, 0.0]), pts - np.array([360.0, 0.0])]
    )
    vals_all = np.concatenate([vals_v, vals_v, vals_v])

    reg = griddata(
        pts_all,
        vals_all,
        (lon2d_reg, lat2d_reg),
        method="linear",
        fill_value=np.nan,
    )
    if np.any(np.isnan(reg)):
        reg_nn = griddata(
            pts_all,
            vals_all,
            (lon2d_reg, lat2d_reg),
            method="nearest",
        )
        reg = np.where(np.isnan(reg), reg_nn, reg)
    return reg

## Summary

The analysis compares the 200–1000 m nitrate-to-phosphate stoichiometry in WOA23 and three CMIP6 historical models: CESM2, GFDL-ESM4 and MPI-ESM1-2-LR.

### Methodology actually used

- **WOA23:** used the annual objectively analysed nitrate and phosphate fields (`n_an`, `p_an`) from the all-period WOA23 files. Values with magnitude ≥1e19 were set to missing. The 200–1000 m layer value at each horizontal cell was a thickness-weighted mean over the standard depth levels whose centres lie in 200–1000 m, using the WOA depth bounds. Horizontal volume weights for statistics used \(\cos(lat)\times \Delta lon \times \Delta z\), with each field’s own valid mask and with N:P computed only where both nitrate and phosphate were valid.
- **CMIP6:** used historical monthly `no3` and `po4` for 1971–2000 inclusive. For each model, each calendar month was averaged across years, then the 12 monthly climatologies were averaged to an annual-mean state, without day weighting. Concentrations were converted from mmol m\(^{-3}\) to µmol kg\(^{-1}\) using the stated constant density \(\rho = 1025\) kg m\(^{-3}\): \(x_{\mu mol\,kg^{-1}} = x_{mmol\,m^{-3}}\times 1000/1025\). The 200–1000 m layer mean was formed as \(\sum(x\cdot \mathrm{volcello})/\sum(\mathrm{volcello})\) over model levels with centres in 200–1000 m; `volcello` supplied all volume weights, and ocean points were defined from each field’s valid cells/`deptho`.
- **Statistics:** N:P was computed from the climatological-mean nitrate and phosphate fields, not as a time average of instantaneous ratios. The headline mean N:P is the volume-weighted mean of cell-by-cell N:P. The second mean is the ratio of the volume-weighted mean nitrate to the volume-weighted mean phosphate. The OLS regression \(\mathrm{NO3}=a\,\mathrm{PO4}+b\) used equal cell weights over all valid cells, as requested; the means and exceedance percentages were volume-weighted.
- **Basins:** WOA23 `basinmask_01` was read; the surface label was used throughout the layer (`depth_dependent=False`). Marginal-sea codes were folded into parent oceans via the code’s centroid-based classifier. Southern and Arctic were kept separate and excluded from named-ocean basin means. For CMIP6, the WOA basin grid was sampled nearest-neighbour at model cell centres.
- **Missing data:** 1e20-style fill values were decoded before statistics; missing cells were excluded from both numerator and denominator in exceedance percentages.

## Interpretation

- **All four sources sit below Redfield overall.** WOA23 has a headline mean N:P of 14.21 and an OLS slope of 12.42, both below 16. The three models also have headline means below 16 (11.75–14.05) and fitted slopes below 16 (2.39–11.10). None is at or above Redfield in the 200–1000 m global mean.
- **WOA23 observed structure:** the observed global low-N:P volume is modest: only 1.9% of volume has N:P<12, with 2.1% in the Pacific and 4.4% in the Indian Ocean. The Atlantic is closest to Redfield (15.47), while the Pacific and Indian are lower (13.71 and 13.84). The observed low-N:P denitrification signature is therefore present mainly in the Pacific and Indian, but it involves only a small fraction of the 200–1000 m volume.
- **GFDL-ESM4 is closest to WOA23 in the global mean and regression.** Its headline mean is 14.05, slope 11.10 and \(R^2=0.809\), and 70.0% of its volume lies between N:P 14 and 18 versus 52.0% for WOA23. However, its low-N:P volume is too large globally (8.0% vs 1.9%) and in the Pacific (17.2% vs 2.1%), while its Indian low-N:P volume is essentially absent (0.3% vs 4.4%). Thus GFDL-ESM4 broadly matches the near-Redfield global stoichiometry but does not reproduce the observed Indian low-N:P signature.
- **CESM2 strongly overproduces low-N:P water.** Its global mean is 11.75 and its slope only 2.39 with \(R^2=0.061\), indicating a very weak linear NO3–PO4 relationship in the climatological mean. It has 38.1% of global volume below N:P 12, with 60.8% in the Pacific and 40.1% in the Indian. This is far larger than observed, although the sign of the low-N:P signal is in the right basins.
- **MPI-ESM1-2-LR also overproduces low-N:P water.** Its mean is 12.35, slope 7.98 and \(R^2=0.653\). The global N:P<12 volume is 21.7%, with 39.7% in the Pacific and 11.1% in the Indian. It qualitatively places enhanced low N:P in the Pacific and Indian, but the amplitude is much larger than WOA23. It has almost no volume between N:P 14 and 18 (0.5%), so its distribution is strongly skewed away from Redfield compared with the observed 52.0%.
- **Cross-model spread is consistent with the different N:P formulations.** GFDL-ESM4’s fixed N:P per plankton type produces a narrower distribution closer to Redfield and misses the Indian low-N:P volume. MPI-ESM1-2-LR’s prescribed Redfield formulation produces a lower mean and too much low-N:P volume. CESM2’s variable C:P formulation produces a very broad, weakly correlated NO3–PO4 distribution and dramatically overestimates low-N:P volume. None of the models quantitatively reproduces the observed low-N:P volume and basin distribution, although CESM2 and MPI-ESM1-2-LR at least place the strongest low-N:P signal in the Pacific/Indian rather than the Atlantic.
