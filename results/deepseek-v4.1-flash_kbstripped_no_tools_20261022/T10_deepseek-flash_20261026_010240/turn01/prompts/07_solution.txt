def get_depth_bounds(ds):
    if "depth_bnds" in ds:
        b = ds["depth_bnds"].values
        if b.shape[0] == 2 and b.shape[1] == len(ds["depth"]):
            b = b.T
        return b
    elif "depth" in ds and "bounds" in ds["depth"].attrs:
        bname = ds["depth"].attrs["bounds"]
        b = ds[bname].values
        if b.shape[0] == 2 and b.shape[1] == len(ds["depth"]):
            b = b.T
        return b
    else:
        d = ds["depth"].values
        edges = np.zeros(len(d) + 1)
        edges[1:-1] = 0.5 * (d[:-1] + d[1:])
        edges[0] = 0.0
        edges[-1] = d[-1] + 0.5 * (d[-1] - d[-2])
        return np.column_stack([edges[:-1], edges[1:]])


def read_msk(path):
    """
    Robustly read a WOA .msk text table into a ragged list of float rows.

    The header of these files varies (comment lines, an optional '#', column
    names containing 'Latitude'/'Longitude').  Instead of trusting a fixed
    header offset, every line is scanned for numeric tokens; the first two
    tokens (assumed Latitude/Longitude) must be plausible geographic
    coordinates.  Empty fields (blanks, or levels below the sea floor) simply
    contribute no tokens, which is what we want since we take the first valid
    value downstream.
    """
    rows = []
    with open(path) as f:
        for line in f:
            s = line.strip()
            if not s or s.startswith("#"):
                continue
            vals = []
            for tok in s.split():
                try:
                    vals.append(float(tok.lstrip("+")))
                except ValueError:
                    continue
            if len(vals) < 3:
                continue
            la, lo = vals[0], vals[1]
            if not (-90.0 <= la <= 90.0 and -180.0 <= lo <= 180.0):
                continue
            rows.append(vals)
    return rows

def sph_xyz(lat, lon):
    lat = np.deg2rad(lat)
    lon = np.deg2rad(lon)
    x = np.cos(lat) * np.cos(lon)
    y = np.cos(lat) * np.sin(lon)
    z = np.sin(lat)
    return np.column_stack([x, y, z])


def rho_insitu(T, S, P):
    """UNESCO 1983 (EOS-80) in-situ density (kg m-3). T degC, S PSU, P dbar."""
    T = np.asarray(T)
    S = np.asarray(S)
    P = np.asarray(P)

    rho_w = (999.842594 + 6.793952e-2 * T - 9.095290e-3 * T**2
             + 1.001685e-4 * T**3 - 1.120083e-6 * T**4 + 6.536332e-9 * T**5)
    A = (8.24493e-1 - 4.0899e-3 * T + 7.6438e-5 * T**2
         - 8.2467e-7 * T**3 + 5.3875e-9 * T**4)
    B = -5.72466e-3 + 1.0227e-4 * T - 1.6546e-6 * T**2
    C = 4.8314e-4
    rho_0 = rho_w + A * S + B * S**1.5 + C * S**2

    K_w = (19652.21 + 148.4206 * T - 2.327105 * T**2
           + 1.360477e-2 * T**3 - 5.155288e-5 * T**4)
    K_0 = (K_w
           + (54.6746 - 0.603459 * T + 1.09987e-2 * T**2 - 6.1670e-5 * T**3) * S
           + (7.944e-2 + 1.6483e-2 * T - 5.3009e-4 * T**2) * S**1.5)

    A_K = 3.239908 + 1.43713e-3 * T + 1.16092e-4 * T**2 - 5.77905e-7 * T**3
    B_K = 2.2838e-3 - 1.0981e-5 * T - 1.6078e-6 * T**2
    C_K = 1.91075e-4
    D_K = 8.50935e-5 - 6.12293e-6 * T + 5.2787e-8 * T**2
    E_K = -9.9348e-7 + 2.0816e-8 * T + 9.1697e-10 * T**2

    K = (K_0
         + P * (A_K + B_K * S + C_K * S**1.5)
         + P**2 * (D_K + E_K * S))
    return rho_0 / (1 - P / K)


def parent_basin(code, lat, lon):
    if code in [1, 2, 3, 4, 5]:
        return code
    # WOA13 marginal-sea groupings
    if code in [6, 7, 8, 11, 12, 13]:
        return 1  # Atlantic
    if code in [9, 10, 18, 19, 20]:
        return 3  # Indian
    if code in [14, 15, 16, 17]:
        return 2  # Pacific
    # Geographic fallback
    if (-10 <= lon <= 45) and (30 <= lat <= 70):
        return 1
    if (30 <= lon <= 100) and (-10 <= lat <= 30):
        return 3
    if ((100 <= lon <= 180) or (-180 <= lon <= -100)) and (0 <= lat <= 66):
        return 2
    if (-100 <= lon <= -50) and (10 <= lat <= 70):
        return 1
    if lat > 66:
        return 4
    if lat < -60:
        return 5
    if -100 <= lon <= 20:
        return 1
    if 20 < lon <= 120:
        return 3
    return 2


def regrid_to_woa(lat2d, lon2d, field):
    """Nearest/linear regrid of a curvilinear field to the WOA 1-degree grid."""
    mask = np.isfinite(field)
    if mask.sum() == 0:
        return np.full((len(lat_woa), len(lon_woa)), np.nan)

    lon360 = lon2d % 360
    lat_s = lat2d[mask]
    lon_s = lon360[mask]
    val_s = field[mask]

    lon_s_wrap = np.concatenate([lon_s, lon_s + 360, lon_s - 360])
    lat_s_wrap = np.concatenate([lat_s, lat_s, lat_s])
    val_s_wrap = np.concatenate([val_s, val_s, val_s])

    lon_t = lon_woa % 360
    lat_t = lat_woa
    lon_t_grid, lat_t_grid = np.meshgrid(lon_t, lat_t)

    out = griddata((lon_s_wrap, lat_s_wrap), val_s_wrap,
                   (lon_t_grid, lat_t_grid), method="linear")
    if np.isnan(out).any():
        out_near = griddata((lon_s_wrap, lat_s_wrap), val_s_wrap,
                            (lon_t_grid, lat_t_grid), method="nearest")
        out = np.where(np.isnan(out), out_near, out)
    return out

## 1. What went wrong

The error is a NumPy `IndexError` raised by indexing a **0-dimensional array with three indices** (a `[:, None, None]`-style expansion of something that has no dimension left). In the executed code there is exactly one expression that expands a one-dimensional vertical coordinate this way and is guaranteed to be reached for every model:

```python

p = depth_m[:, None, None] * np.ones_like(o2_native.values)
```

`depth_m` must be a 1-D array over the model's native levels. If `o2_native.lev` is missing, has been renamed, or — more likely here — survives `xr.align(o2_native, thetao, so, volcello, areacello, join="inner")` only as a **scalar/0-D coordinate**, then `depth_m` is 0-D and `depth_m[:, None, None]` raises precisely the message observed. Three model/experiment pairs (`GFDL-ESM4` and `MPI-ESM1-2-LR` historical, then the three ssp245 runs) all pass through this line, so the failure occurs before any diagnostic is computed, which is why nothing at all was reported.

Two related fragilities in the same code path make this the most plausible location:

- **Variable lookup by substring.** `get_period_mean` first tries `var in ds`, otherwise takes the first data variable whose lower-case name *contains* the requested string. For `so` this can silently select a surface field (`sos`, `soga`, …) with 2-D `(lat, lon)` dimensions. That would not itself raise this error (2-D salinity broadcasts against the 3-D temperature), but it would leave `so` with no vertical dimension and can trigger dimension mismatches downstream. The same lookup is used for `volcello`, `areacello` and `deptho`, where a wrongly chosen variable can collapse the vertical axis.

For CESM2 the vertical coordinate is also stored in centimetres (`depth_scale = 0.01`), so any coordinate loss or unit mix-up there propagates directly into pressure, density and hence into the µmol kg⁻¹ conversion.
