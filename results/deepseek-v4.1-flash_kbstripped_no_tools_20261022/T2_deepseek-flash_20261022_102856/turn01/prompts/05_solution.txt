2. **Build RECON_9120.** From the ERSSTv5 field alone, compute a monthly climatology for 1991-2020 (one value per calendar month, per grid cell), subtract it month-by-month to get the SST anomaly, then average the anomaly over the Nino 3.4 box 5°S–5°N, 170°W–120°W with cos(latitude) area weights.
   - Note [omip-weighted-means-and-integrals]: this regional average is physical, so use area weights (cos lat), not a plain mean over the box, and restrict the weights to exactly the cells that are valid in the data so land/missing cells stay out of the denominator.
   - Note [anomaly-baselines-and-climate-indices]: the anomaly must be built from ERSSTv5's *own* 1991-2020 monthly climatology, matched month to calendar month (which removes the seasonal cycle); record the climatology and baseline used.    The result is the monthly RECON_9120 series, 1950-01 to 2023-12.

if lat_vals[0] > lat_vals[-1]:

def build_nino34_index(sst_da, clim_start, clim_end, name):
    """Monthly Nino 3.4 index from SST, using a fixed monthly climatology."""
    clim = (
        sst_da.sel(time=slice(clim_start, clim_end))
        .groupby("time.month")
        .mean(dim="time")
    )
    anom = sst_da.groupby("time.month") - clim

    # cos(latitude) area weights, restricted to cells valid in the data.
    w = np.cos(np.deg2rad(sst_da["lat"]))
    valid = np.isfinite(anom)
    num = (anom * w).where(valid).sum(dim=("lat", "lon"), skipna=True)
    den = w.where(valid).sum(dim=("lat", "lon"), skipna=True)
    idx = (num / den).rename(name)
    return idx.compute()

def read_cpc_nino_monthly(path):
    rows = []
    with open(path, "r") as f:
        for line in f:
            parts = line.strip().split()
            if len(parts) < 10:
                continue
            try:
                yr = int(parts[0])
                mon = int(parts[1])
            except ValueError:
                continue
            vals = []
            ok = True
            for x in parts[2:10]:
                try:
                    vals.append(float(x))
                except ValueError:
                    ok = False
                    break
            if not ok:
                continue
            rows.append([yr, mon] + vals)

    cols = [
        "YR", "MON",
        "NINO1+2", "NINO1+2_ANOM",
        "NINO3", "NINO3_ANOM",
        "NINO4", "NINO4_ANOM",
        "NINO3.4", "NINO3.4_ANOM",
    ]
    df = pd.DataFrame(rows, columns=cols)

    for c in cols[2:]:
        df[c] = pd.to_numeric(df[c], errors="coerce")
        # CPC missing-value sentinels are around -999; real anomalies are O(1).
        df.loc[df[c] <= -90, c] = np.nan

    df["time"] = pd.to_datetime(dict(year=df["YR"], month=df["MON"], day=1))
    return df

def read_oni(path):
    rows = []
    with open(path, "r") as f:

            parts = line.strip().split()
            if len(parts) < 4:

                [seas, yr, total, anom,
                 pd.Timestamp(year=yr, month=month, day=1)]
            )
    return pd.DataFrame(

def detect_threshold_runs(series, threshold, min_duration=5, direction="high"):
    """Find contiguous runs of at least min_duration months beyond threshold."""
    vals = series.values.astype(float)
    times = pd.to_datetime(series.index)

    if direction == "high":
        cond = np.isfinite(vals) & (vals >= threshold)
    else:
        cond = np.isfinite(vals) & (vals <= threshold)

    events = []
    i = 0
    n = len(vals)
    while i < n:
        if cond[i]:
            j = i
            while j + 1 < n and cond[j + 1]:
                j += 1
            length = j - i + 1
            if length >= min_duration:
                seg = vals[i:j + 1]
                if direction == "high":
                    k = i + int(np.nanargmax(seg))
                else:
                    k = i + int(np.nanargmin(seg))
                events.append(
                    {
                        "start": times[i],
                        "end": times[j],
                        "duration": length,
                        "peak": float(vals[k]),
                        "peak_time": times[k],
                    }
                )
            i = j + 1
        else:
            i += 1
    return events
