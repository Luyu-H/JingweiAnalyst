def lonlat_to_xyz(lat, lon):
    lat = np.deg2rad(np.asarray(lat, dtype=float))
    lon = np.deg2rad(np.asarray(lon, dtype=float))
    return np.column_stack(
        [np.cos(lat) * np.cos(lon), np.cos(lat) * np.sin(lon), np.sin(lat)]
    )


def make_regrid_indices(src_lat, src_lon, tgt_lat, tgt_lon):
    src_lat = np.asarray(src_lat)
    src_lon = np.asarray(src_lon)
    if src_lat.ndim == 1 and src_lon.ndim == 1:
        src_lon2, src_lat2 = np.meshgrid(src_lon, src_lat)
    else:
        src_lat2, src_lon2 = src_lat, src_lon
    tgt_lat2, tgt_lon2 = np.meshgrid(tgt_lat, tgt_lon, indexing="ij")
    src_xyz = lonlat_to_xyz(src_lat2.ravel(), src_lon2.ravel())
    tgt_xyz = lonlat_to_xyz(tgt_lat2.ravel(), tgt_lon2.ravel())
    tree = cKDTree(src_xyz)
    _, idx = tree.query(tgt_xyz, k=1)
    return idx.reshape(len(tgt_lat), len(tgt_lon))


def regrid_array(arr, idx):
    arr = np.asarray(arr)
    lead = arr.shape[:-2]
    nsp = arr.shape[-2] * arr.shape[-1]
    flat = arr.reshape(-1, nsp)
    out = flat[:, idx.ravel()]
    return out.reshape(lead + idx.shape)


def weighted_corr(x, y, w):
    x = np.asarray(x).ravel()
    y = np.asarray(y).ravel()
    w = np.asarray(w).ravel()
    valid = np.isfinite(x) & np.isfinite(y) & np.isfinite(w)
    x, y, w = x[valid], y[valid], w[valid]
    if len(x) < 2:
        return np.nan
    wsum = w.sum()
    mx = np.sum(w * x) / wsum
    my = np.sum(w * y) / wsum
    cov = np.sum(w * (x - mx) * (y - my)) / wsum
    vx = np.sum(w * (x - mx) ** 2) / wsum
    vy = np.sum(w * (y - my) ** 2) / wsum
    if vx <= 0 or vy <= 0:
        return np.nan
    return cov / np.sqrt(vx * vy)


def linear_trend_ess(x, y):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    valid = np.isfinite(x) & np.isfinite(y)
    x, y = x[valid], y[valid]
    n = len(x)
    if n < 3:
        return dict(
            slope=np.nan, intercept=np.nan, ci_low=np.nan, ci_high=np.nan,
            r2=np.nan, excludes_zero=False, p_value=np.nan, n=n,
            n_eff=np.nan, rho=np.nan,
        )
    x_mean = x.mean()
    y_mean = y.mean()
    Sxx = np.sum((x - x_mean) ** 2)
    Sxy = np.sum((x - x_mean) * (y - y_mean))
    slope = Sxy / Sxx
    intercept = y_mean - slope * x_mean
    resid = y - (intercept + slope * x)
    sse = np.sum(resid ** 2)

    denom = np.sqrt(np.sum(resid[:-1] ** 2) * np.sum(resid[1:] ** 2))
    rho = np.sum(resid[:-1] * resid[1:]) / denom if denom > 0 else 0.0
    rho = float(np.clip(rho, -0.99, 0.99))
    n_eff = n * (1 - rho) / (1 + rho)
    n_eff = float(np.clip(n_eff, 2.0, n))

    df_base = max(n - 2, 1)
    sigma2 = sse / df_base
    se = np.sqrt(sigma2 / Sxx * (n / n_eff))
    df = max(n_eff - 2.0, 1.0)
    tcrit = stats.t.ppf(0.975, df)
    ci = tcrit * se

    syy = np.sum((y - y_mean) ** 2)
    r2 = 1.0 - sse / syy if syy > 0 else np.nan
    pval = 2.0 * stats.t.sf(abs(slope / se), df) if se > 0 else np.nan
    excludes_zero = bool((slope - ci > 0) or (slope + ci < 0))

    return dict(
        slope=float(slope),
        intercept=float(intercept),
        ci_low=float(slope - ci),
        ci_high=float(slope + ci),
        r2=float(r2),
        excludes_zero=excludes_zero,
        p_value=float(pval),
        n=int(n),
        n_eff=float(n_eff),
        rho=float(rho),
    )


def trend_map_ess(annual_da):
    years = annual_da["year"].values.astype(float)
    spatial_dims = [d for d in annual_da.dims if d != "year"]
    arr = annual_da.transpose("year", *spatial_dims).values
    n = arr.shape[0]
    y = arr.reshape(n, -1)
    x = years - years.mean()
    Sxx = np.sum(x ** 2)

    valid = np.isfinite(y)
    count = valid.sum(axis=0)
    y_sum = np.nansum(y, axis=0)
    y_mean = np.divide(
        y_sum, count,
        out=np.full_like(y_sum, np.nan, dtype=float),
        where=count > 0,
    )

    y_centered = y - y_mean[None, :]
    Sxy = np.nansum(x[:, None] * y_centered, axis=0)
    slope = Sxy / Sxx
    intercept = y_mean - slope * years.mean()
    resid = y - (intercept[None, :] + slope[None, :] * years[:, None])
    sse = np.nansum(resid ** 2, axis=0)

    num = np.nansum(resid[:-1, :] * resid[1:, :], axis=0)
    den1 = np.nansum(resid[:-1, :] ** 2, axis=0)
    den2 = np.nansum(resid[1:, :] ** 2, axis=0)
    denom = np.sqrt(den1 * den2)
    rho = np.divide(num, denom, out=np.zeros_like(num), where=denom > 0)
    rho = np.clip(rho, -0.99, 0.99)
    n_eff = n * (1 - rho) / (1 + rho)
    n_eff = np.clip(n_eff, 2, n)

    df_base = np.maximum(count - 2, 1)
    sigma2 = sse / df_base
    sigma2 = np.where(count >= 3, sigma2, np.nan)
    se = np.sqrt(sigma2 / Sxx * (n / n_eff))
    df = np.maximum(n_eff - 2, 1)
    pval = 2.0 * stats.t.sf(np.abs(slope / se), df)
    slope = np.where(count >= 3, slope, np.nan)
    pval = np.where(count >= 3, pval, np.nan)

    return slope.reshape(arr.shape[1:]), pval.reshape(arr.shape[1:])


def open_so_files(model, experiment, root):
    d = os.path.join(root, model, "so")
    files = sorted(glob.glob(os.path.join(d, "so_*.nc")))
    if not files:
        raise FileNotFoundError(f"No so files found for {model} {experiment} under {d}")
    variants = {}
    for f in files:
        m = re.search(r"_(r\d+i\d+p\d+f\d+)_", os.path.basename(f))
        key = m.group(1) if m else "unknown"
        variants.setdefault(key, []).append(f)
    variant = max(variants, key=lambda k: len(variants[k]))
    files = variants[variant]
    ds = xr.open_mfdataset(
        files, combine="by_coords", use_cftime=True, decode_times=True
    )
    return ds, variant, files


def open_fixed_field(model, var, root, fallback_root=None, table="Ofx"):
    dirs = [os.path.join(root, model, var)]
    if fallback_root is not None:
        dirs.append(os.path.join(fallback_root, model, var))
    for d in dirs:

            continue

            files = tab_files

        # GFDL volcello is Omon and split in time; one file already has the full grid.

    return None


def grid_from_points(lat_vals, lon_vals, vals, tgt_lat, tgt_lon):
    grid = np.full((len(tgt_lat), len(tgt_lon)), np.nan, dtype=float)
    dlat = tgt_lat[1] - tgt_lat[0]
    dlon = tgt_lon[1] - tgt_lon[0]
    ilat = np.round((np.asarray(lat_vals) - tgt_lat[0]) / dlat).astype(int)
    ilon = np.round((np.asarray(lon_vals) - tgt_lon[0]) / dlon).astype(int)
    ok = (
        (ilat >= 0) & (ilat < len(tgt_lat))
        & (ilon >= 0) & (ilon < len(tgt_lon))
    )
    grid[ilat[ok], ilon[ok]] = np.asarray(vals)[ok]
    return grid


def read_woa_msk(path):
    """Read a WOA .msk comma-separated table robustly.

    The basin mask is a sparse table with one row per ocean point and one basin
    number per standard level; levels below the sea floor are stored as empty
    fields.  ``np.loadtxt`` cannot parse these empty strings, so we parse line
    by line, convert empty fields to NaN, and pad ragged rows with NaN.
    """
    rows = []
    maxlen = 0
    with open(path) as fh:
        fh.readline()  # header / comment row 1
        fh.readline()  # header / comment row 2
        for line in fh:
            s = line.strip().rstrip(",")
            if not s:
                continue
            fields = [x.strip() for x in s.split(",")]
            vals = []
            ok = True
            for x in fields:
                if x == "":
                    vals.append(np.nan)
                else:
                    try:
                        vals.append(float(x))
                    except ValueError:
                        ok = False
                        break
            if not ok or len(vals) < 2:
                continue
            rows.append(vals)
            if len(vals) > maxlen:
                maxlen = len(vals)
    arr = np.full((len(rows), maxlen), np.nan, dtype=float)
    for i, r in enumerate(rows):
        arr[i, :len(r)] = r
    return arr


def fold_basin_mask(basin_grid, lat, lon):
    major_codes = np.array([1, 2, 3, 4, 5])
    folded = np.full_like(basin_grid, np.nan, dtype=float)

    for c in major_codes:
        folded[basin_grid == c] = c

    lat2d, lon2d = np.meshgrid(lat, lon, indexing="ij")
    major_mask = np.isin(basin_grid, major_codes)
    if not major_mask.any():
        return folded

    major_lat = lat2d[major_mask]
    major_lon = lon2d[major_mask]
    major_vals = basin_grid[major_mask]
    tree = cKDTree(lonlat_to_xyz(major_lat, major_lon))

    high_codes = [
        c for c in np.unique(basin_grid)
        if np.isfinite(c) and c > 0 and c not in major_codes
    ]
    for c in high_codes:
        mask_c = basin_grid == c
        if not mask_c.any():
            continue
        clat = lat2d[mask_c]
        clon = lon2d[mask_c]
        _, nearest_idx = tree.query(lonlat_to_xyz(clat, clon), k=1)
        nearest_major = major_vals[nearest_idx]
        vals, counts = np.unique(nearest_major, return_counts=True)
        parent = vals[np.argmax(counts)]
        folded[mask_c] = parent
    return folded

## Methodology Summary

- **WOA23 reference:** The objectively analysed salinity field `s_an` was read from the 1971–2000 climate normal file, using time index 0. Levels at 0, 5 and 10 m were selected. Level weights were full cell volumes computed as WOA analytic 1° cell areas multiplied by full level thicknesses from `depth_bnds`, with no clipping to the 0–10 m interval. SSS was the volume-weighted mean over those levels. The land–sea mask used `landsea_01 > 1`. The basin mask used the surface WOA basin labels; higher-number marginal labels were folded to the nearest major basin. Salinity is practical salinity in psu.
- **CMIP6 processing:** Monthly `so` was taken for historical 1950–2014 and ssp245 2071–2100 for CESM2, GFDL‑ESM4 and MPI‑ESM1‑2‑LR. Model levels within 0–10 m were selected using the model level coordinate, with CESM2 levels converted from cm to m (`depth_scale=0.01`). Each selected level was weighted by its full `volcello` cell volume, without clipping to the 0–10 m overlap. Monthly SSS was formed and then averaged to annual means. Each model’s own `areacello` and `volcello` were opened once and used for both experiments; non-positive areas/volumes were masked, and CESM2 additionally used `sftof > 0`.
- **Regridding and comparison mask:** Model fields were mapped to the WOA23 1° longitude–latitude grid with a nearest-neighbour index built on Cartesian coordinates on the sphere. The common comparison mask required WOA `landsea_01 > 1`, valid WOA SSS, and valid regridded 1971–2000 SSS for all three models. It contains **40,778 cells**. Bias, RMSE, pattern correlation and comparison global means were computed only on this common mask, with area weighting: regridded model `areacello` for models and WOA analytic cell areas for WOA.
- **Trends:** Annual global-mean SSS was computed on each model’s native grid using native `areacello`. Separate linear trends were fitted for historical 1950–2014 and ssp245 2071–2100. The 95% confidence intervals use an effective-sample-size correction based on the lag-1 autocorrelation of residuals. Cell-by-cell historical trends were also computed with the same correction and p-values for Figure 2. No trend was fitted across the 2015–2070 gap.
- **Delta SSS:** Delta was ssp245 2071–2100 mean minus historical 1995–2014 mean, on cells valid in both periods. Global and basin means were area-weighted with the model’s regridded `areacello`. Basins were defined from the WOA23 basin mask after folding marginal seas into their parent ocean; reported basins are Atlantic (code 1), Pacific (code 2) and Indian (code 3).
