def get_variant(model):
    grid = models[model]["grid_label"]
    pattern = os.path.join(
        root, model, "thetao",
        f"thetao_Omon_{model}_historical_*_{grid}*.nc"
    )
    files = sorted(glob.glob(pattern))
    if not files:
        raise FileNotFoundError(f"No thetao files found for {model}: {pattern}")
    variants = []
    for f in files:
        parts = os.path.basename(f).split("_")
        if len(parts) >= 5:
            variants.append(parts[4])
    if not variants:
        raise RuntimeError(f"Could not parse variant from {files}")
    return sorted(set(variants))[0]

# Operator MLD from thetao/so
# -----------------------------------------------------------------------------
def compute_operator_mld(thetao_t, so_t, lat2d, lon2d, z, threshold=0.03):
    """
    Compute mixed-layer depth from a single monthly thetao/so profile.

    Parameters
    ----------
    thetao_t, so_t : ndarray (nlev, ny, nx)
        Potential temperature and practical salinity.
    lat2d, lon2d : ndarray (ny, nx)
    z : ndarray (nlev,)
        Positive-down model level depths in metres.

    Returns
    -------
    mld : ndarray (ny, nx)
        Operator MLD, NaN where no crossing is found.
    no_cross : ndarray (ny, nx)
        True where the surface profile is valid but the threshold is never
        reached.
    """
    # Treat unphysically large values as fill.
    th = np.where(np.isfinite(thetao_t) & (np.abs(thetao_t) < 1e10), thetao_t, np.nan)
    sp = np.where(np.isfinite(so_t) & (np.abs(so_t) < 1e10), so_t, np.nan)

    nlev, ny, nx = th.shape

    # Pressure and TEOS-10 conversions.
    z3 = np.broadcast_to(-z[:, None, None], (nlev, ny, nx))
    lat3 = np.broadcast_to(lat2d[None, :, :], (nlev, ny, nx))
    lon3 = np.broadcast_to(lon2d[None, :, :], (nlev, ny, nx))

    p = gsw.p_from_z(z3, lat3)
    SA = gsw.SA_from_SP(sp, p, lon3, lat3)
    CT = gsw.CT_from_pt(SA, th)
    sig0 = gsw.sigma0(SA, CT)

    # Reference = shallowest model level.
    ref = sig0[0, :, :]
    diff = sig0 - ref[None, :, :]

    finite = np.isfinite(diff)
    cond = (diff >= threshold) & finite

    # Do not cross a missing level: require no NaNs before the candidate level.
    cum_nan = np.cumsum(~finite, axis=0)
    cond = cond & (cum_nan == 0)

    has = cond.any(axis=0)
    k = np.argmax(cond, axis=0)
    has = has & (k > 0)

    idx_prev = np.maximum(k - 1, 0)

    sig_prev = np.take_along_axis(sig0, idx_prev[None, :, :], axis=0)[0]
    sig_k = np.take_along_axis(sig0, k[None, :, :], axis=0)[0]

    z_prev = z[idx_prev]
    z_k = z[k]

    denom = sig_k - sig_prev
    frac = np.where(denom != 0.0, (threshold - (sig_prev - ref)) / denom, 0.0)
    mld = z_prev + frac * (z_k - z_prev)
    mld = np.where(has, mld, np.nan)

    surface_valid = np.isfinite(th[0]) & np.isfinite(sp[0])
    no_cross = surface_valid & ~has

    return mld, no_cross

def weighted_pearson(x, y, w):
    sw = w.sum()
    if sw <= 0:
        return np.nan
    xm = (w * x).sum() / sw
    ym = (w * y).sum() / sw
    cov = (w * (x - xm) * (y - ym)).sum() / sw
    vx = (w * (x - xm) ** 2).sum() / sw
    vy = (w * (y - ym) ** 2).sum() / sw
    if vx <= 0 or vy <= 0:
        return np.nan
    return cov / np.sqrt(vx * vy)


def compute_metrics(op_clim, ml_clim, area, total_ocean_area):
    mask = build_comparison_mask(op_clim, ml_clim, ocean_only=False)
    op_m = op_clim.where(mask)
    ml_m = ml_clim.where(mask)
    D = (op_m - ml_m).where(mask)
    w = area.where(mask)

    sw = float(w.sum())
    mean = float((D * w).sum() / sw)
    rmse = float(np.sqrt((D ** 2 * w).sum() / sw))

    x = op_m.values.ravel()
    y = ml_m.values.ravel()
    ww = w.values.ravel()
    valid = np.isfinite(x) & np.isfinite(y) & np.isfinite(ww) & (ww > 0)
    corr = weighted_pearson(x[valid], y[valid], ww[valid])

    exceed_area = float(w.where(np.abs(D) > 20.0).sum())
    exceed_pct = 100.0 * exceed_area / sw
    coverage_pct = 100.0 * sw / total_ocean_area
    n_cells = int(mask.sum())

    metrics = {
        "mean": mean,
        "rmse": rmse,
        "corr": corr,
        "exceed_pct": exceed_pct,
        "coverage_pct": coverage_pct,
        "n_cells": n_cells,
        "common_area": sw,
    }
    return metrics, mask, D, op_m, ml_m


def regional_stats(D, area, lat2d):
    bands = [
        (-90.0, -60.0),
        (-60.0, -40.0),
        (-40.0, -20.0),
        (-20.0, 0.0),
        (0.0, 20.0),
        (20.0, 40.0),
        (40.0, 60.0),
        (60.0, 90.0),
    ]
    out = []
    for lo, hi in bands:
        m = (lat2d >= lo) & (lat2d < hi)
        w = area.where(m)
        d = D.where(m)
        sw = float(w.sum())
        if sw > 0:
            mean = float((d * w).sum() / sw)
            rmse = float(np.sqrt((d ** 2 * w).sum() / sw))
        else:
            mean = np.nan
            rmse = np.nan
        out.append({"lat_lo": lo, "lat_hi": hi, "mean": mean, "rmse": rmse, "area": sw})
    return out

def remap_conservative(field, area, lat2d, lon2d, dlon=2.0, dlat=2.0):
    lon = ((lon2d + 180.0) % 360.0) - 180.0
    lat = lat2d

    lon_edges = np.arange(-180.0, 180.0 + dlon, dlon)
    lat_edges = np.arange(-90.0, 90.0 + dlat, dlat)
    nlon = len(lon_edges) - 1
    nlat = len(lat_edges) - 1

    ix = np.clip(np.floor((lon - lon_edges[0]) / dlon).astype(int), 0, nlon - 1)
    iy = np.clip(np.floor((lat - lat_edges[0]) / dlat).astype(int), 0, nlat - 1)

    valid = np.isfinite(field) & np.isfinite(area) & (area > 0)
    flat_idx = iy[valid] * nlon + ix[valid]

    sum_w = np.bincount(
        flat_idx, weights=area[valid], minlength=nlat * nlon
    ).reshape(nlat, nlon)
    sum_fw = np.bincount(
        flat_idx, weights=area[valid] * field[valid], minlength=nlat * nlon
    ).reshape(nlat, nlon)

    out = np.full((nlat, nlon), np.nan)
    np.divide(sum_fw, sum_w, out=out, where=sum_w > 0)

    lon_centers = 0.5 * (lon_edges[:-1] + lon_edges[1:])
    lat_centers = 0.5 * (lat_edges[:-1] + lat_edges[1:])

    R = 6371000.0
    target_area = (
        R ** 2
        * np.deg2rad(dlon)
        * (np.sin(np.deg2rad(lat_edges[1:])) - np.sin(np.deg2rad(lat_edges[:-1])))[:, None]
        * np.ones((1, nlon))
    )

    return out, lon_centers, lat_centers, sum_w, target_area

## Direct answer

The offline operator MLD does **not** reproduce the published `mlotst` exactly in any of the six model/month cases, but the level of internal consistency is high for CESM2, intermediate for MPI-ESM1-2-LR, and poorest for GFDL-ESM4. All six cases have a negative area-weighted mean difference, meaning the operator MLD is generally **shallower** than the published `mlotst`. Spatial correlations are high (0.983–0.998), so the main spatial pattern is captured; the disagreements are mainly in bias, root-mean-square difference, and localized high-latitude/midlatitude regions.

## Methodology actually used

- CMIP6 historical monthly `thetao`, `so`, `mlotst`, and fixed `areacello` were opened for CESM2, GFDL-ESM4, and MPI-ESM1-2-LR, all variant `r1i1p1f1`, for **1971-01 through 2000-12**.
- CESM2 area was additionally masked with `sftof > 0`.
- For each model and each of the 30 Marches and 30 Septembers, the operator MLD was computed from that month’s `thetao`/`so` profile using:
  - pressure from model depth and latitude;
  - practical salinity converted to Absolute Salinity;
  - potential temperature converted to Conservative Temperature;
  - potential density anomaly `sigma0`;
  - shallowest model level as reference;
  - threshold `sigma0 - sigma0_ref = 0.03 kg m-3`;
  - linear interpolation of the crossing depth between bracketing levels;
  - missing operator MLD when the threshold was never reached.
- The operator MLD was computed on each monthly profile first, then averaged over the 30 instances. It was not computed from the 30-year mean temperature and salinity.
- For each model/month, the all-time-valid mask required both operator MLD and published `mlotst` to be finite in **all 30** instances of that calendar month. The climatologies and comparisons were made on that common mask.
- Metrics were computed on each model’s native grid with `areacello` area weights. The comparison mask was built with `build_comparison_mask(..., ocean_only=False)` and applied with `.where(mask)`.
- Area-weighted Pearson correlation used `areacello` weights. The exceedance percentage is the area with `|D| > 20 m` divided by common-mask area. Coverage is common-mask area divided by total model ocean area.
- No required model variable was missing; the computation succeeded.

## Why the operator and `mlotst` are not expected to agree exactly

Even though both quantities come from the same simulation, the offline operator differs from the online published `mlotst` diagnostic because:
- the reference level and detailed density criterion may differ;
- the operator uses a `sigma0` difference from the shallowest model level, whereas the model diagnostic may use a different density formulation or a displaced-parcel criterion at local pressure;
- the operator is computed on individual monthly profiles and then averaged, so it is not the same as the model’s online time-mean MLD;
- this time-averaging order tends to make the operator MLD shallower, consistent with the negative mean differences found here;
- non-crossing columns are set to missing, and their area differs by model/month;
- `thetao` masks, `so` masks, and `mlotst` masks can differ, so a common all-time-valid mask is required.

These factors are diagnostic properties, not model physics errors.
