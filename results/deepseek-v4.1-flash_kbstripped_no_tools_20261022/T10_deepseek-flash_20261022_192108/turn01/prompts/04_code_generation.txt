Plan from the previous step:
1. Open the WOA23 files and read the 1971-2000 climate normal fields `oxygen`, `percent_oxygen_saturation` and `apparent_oxygen_utilization` (1° grid, 33-ish standard levels), together with the static masks `landsea_01` (land-sea) and `basinmask_01` (basin). Verify that `oxygen` is already in µmol kg⁻¹ (no density conversion) and that the vertical axis is depth in metres.

2. Open `cmip6_historical` and read `o2`, `thetao`, `so` for GFDL-ESM4 and MPI-ESM1-2-LR, monthly 1971-01 to 2000-12, together with each model's Ofx fixed fields `volcello`, `areacello` and `deptho`. Confirm that CESM2 publishes no `o2` in the historical experiment (record it as missing, do not substitute).

3. Open `cmip6_ssp245` and read `o2`, `thetao`, `so` for CESM2, GFDL-ESM4 and MPI-ESM1-2-LR, monthly 2071-01 to 2100-12, with the same `volcello`, `areacello`, `deptho`. Record the variant labels (GFDL-ESM4 and MPI-ESM1-2-LR ssp245 r1i1p1f1; CESM2 ssp245 r10i1p1f1).

4. For every model/experiment combination, average the monthly `o2`, `thetao` and `so` over its period with `temporal_aggregate(freq='mean')` so that month lengths are weighted, giving one period-mean 3-D field per model and experiment.
   - Note [monthly-means-are-time-averages]: the OMZ diagnostic is applied to the period-mean field (average-then-diagnostic), which is the like-for-like choice against the WOA23 climatology; do not threshold each month and then average, and do not use `compute_basic_stat(op='mean', dim='time')` (equal weight per calendar month).

5. Convert each model's `o2` from mol m⁻³ to µmol kg⁻¹: µmol kg⁻¹ = mmol m⁻³ × 1000/ρ, using the in-situ density computed from the period-mean `thetao` and `so` with `compute_potential_density_anomaly` and `ref_pressure=None`; keep and report `attrs['density_assumption']` with the density actually used.

def files_for(root, model, var):
    d = os.path.join(root, model, var)
    fs = sorted(glob.glob(os.path.join(d, "*.nc")))
    if not fs:
        raise FileNotFoundError(f"no netCDF for {model}/{var} in {d}")
    return fs

def month_lengths(times):
    tv = np.asarray(times)
    n = len(tv)
    if n == 1:
        return np.array([30.0])
    d = np.array([(tv[i + 1] - tv[i]).days for i in range(n - 1)], dtype=float)
    return np.append(d, d[-1])

def period_mean(model, var, root, y0, y1):
    """Month-length weighted mean over the calendar months in [y0, y1]."""
    m = MODEL[model]
    lev, yd, xd = m["lev"], m["ydim"], m["xdim"]
    num, den, ref = None, 0.0, None
    for f in files_for(root, model, var):
        ds = xr.open_dataset(f, use_cftime=True)
        if "time" not in ds.dims:
            ds.close(); continue
        yrs = ds["time"].dt.year.values
        idx = np.where((yrs >= y0) & (yrs <= y1))[0]
        if idx.size == 0:
            ds.close(); continue
        sub = ds.isel(time=idx)
        v = sub[var]
        w = month_lengths(sub["time"].values)
        wda = xr.DataArray(w, dims="time", coords={"time": sub["time"]})
        per_step = v.sizes[lev] * v.sizes[yd] * v.sizes[xd] * 8.0
        nstep = max(1, int(5.0e8 // per_step))
        acc = None
        for k in range(0, idx.size, nstep):
            sl = slice(k, k + nstep)
            c = (v.isel(time=sl) * wda.isel(time=sl)).sum("time", skipna=True).compute()
            acc = c if acc is None else acc + c
        if ref is not None and acc.dims == num.dims:
            acc = acc.assign_coords({dn: ref[dn] for dn in acc.dims})
        else:
            ref = {dn: acc[dn].values.copy() for dn in acc.dims}
        num = acc if num is None else num + acc
        den += float(w.sum())
        ds.close()
    if num is None:
        raise ValueError(f"no months in {y0}-{y1} for {model}/{var}")
    return num / den

def open_static(model, var, roots):
    for root in roots:
        d = os.path.join(root, model, var)
        fs = sorted(glob.glob(os.path.join(d, "*.nc")))
        if not fs:
            continue
        ds = xr.open_dataset(fs[0], use_cftime=True)
        da = ds[var]
        if "time" in da.dims:
            da = da.isel(time=0, drop=True)
        return da.squeeze(drop=True).compute()
    raise FileNotFoundError(f"{var} not found for {model}")

def variant_of(model, var, root):
    fn = os.path.basename(files_for(root, model, var)[0])
    for p in fn.split("_"):
        if re.fullmatch(r"r\d+i\d+p\d+f\d+", p):
            return p
    return None

def rho_sigma0(T, S):
    """EOS-80 potential density at p = 0 (ref_pressure = None), T in degC, S in psu."""
    rw = (999.842594 + 6.793952e-2 * T - 9.095290e-3 * T**2 + 1.001685e-4 * T**3
          - 1.120083e-6 * T**4 + 6.536332e-9 * T**5)
    A = 8.24493e-1 - 4.0899e-3 * T + 7.6438e-5 * T**2 - 8.2467e-7 * T**3 + 5.3875e-9 * T**4
    B = -5.72466e-3 + 1.0227e-4 * T - 1.6546e-6 * T**2
    C = 4.8314e-4
    return rw + A * S + B * S**1.5 + C * S**2

def o2_to_umolkg(da, rho):
    u = str(da.attrs.get("units", "")).lower().replace(" ", "")
    if "kg" in u:                     # already per mass
        return da.assign_attrs(units="umol kg-1",
                               density_assumption="none: field already per mass"),
                 "none"
    if "mmol" in u:
        return (da * 1.0e3 / rho).assign_attrs(units="umol kg-1",
               density_assumption="monthly-mean thetao/so -> EOS-80 sigma0 (ref_pressure=None); umol/kg = mmol/m3 * 1000/rho"), "sigma0"
    return (da * 1.0e6 / rho).assign_attrs(units="umol kg-1",
           density_assumption="monthly-mean thetao/so -> EOS-80 sigma0 (ref_pressure=None); umol/kg = mol/m3 * 1e6/rho"), "sigma0"

def model_latlon(d, model):
    m = MODEL[model]
    la, lo = d[m["lat"]], d[m["lon"]]
    if la.ndim == 1 and lo.ndim == 1:
        LA, LO = np.meshgrid(la.values, lo.values, indexing="ij")
    else:
        LA, LO = np.asarray(la.values), np.asarray(lo.values)
    return LA, LO

def basin_on_model(LA, LO):
    i = np.clip(np.rint(np.asarray(LA, float) + 89.5).astype(int), 0, 179)
    lon180 = ((np.asarray(LO, float) + 180.0) % 360.0) - 180.0
    j = np.rint(lon180 + 179.5).astype(int) % 360
    return basin_full[i, j]

def to_1deg(field, LA, LO):
    from scipy.interpolate import griddata
    f = np.asarray(field, dtype=float)
    lo = np.asarray(LO, dtype=float) % 360.0
    la = np.asarray(LA, dtype=float)
    ok = np.isfinite(f)
    tlon = np.arange(0.5, 360.0, 1.0)
    tlat = np.arange(-89.5, 90.0, 1.0)
    if ok.sum() == 0:
        return tlon, tlat, np.full((len(tlat), len(tlon)), np.nan)
    pts = np.concatenate([np.c_[lo[ok], la[ok]], np.c_[lo[ok] - 360.0, la[ok]], np.c_[lo[ok] + 360.0, la[ok]]])
    val = np.concatenate([f[ok], f[ok], f[ok]])
    TLON, TLAT = np.meshgrid(tlon, tlat)
    out = griddata(pts, val, (TLON, TLAT), method="nearest")
    return tlon, tlat, out
