def detect_events(series, thresh=0.5, min_len=5):
    s = series.dropna().sort_index()
    v = s.values
    n = len(v)
    out = []
    i = 0
    while i < n:
        if v[i] >= thresh:
            j = i
            while j + 1 < n and v[j + 1] >= thresh:
                j += 1
            if (j - i + 1) >= min_len:
                seg = s.iloc[i:j + 1]

                            "peak_time": seg.idxmax(), "peak": float(seg.max()),
                            "n_seasons": int(j - i + 1)})
            i = j + 1
        else:
            i += 1
    return out
