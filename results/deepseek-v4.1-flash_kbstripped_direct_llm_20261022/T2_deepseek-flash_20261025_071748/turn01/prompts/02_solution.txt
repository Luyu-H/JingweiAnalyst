The user's question:
Rebuild the Nino 3.4 index from the raw ERSSTv5 sea surface temperature field and check it against the series NOAA/CPC publishes, then use the same code to reproduce the offset between CPC's two baseline periods.

Data to use:
- ERSSTv5 monthly SST, 1950-01 to 2023-12, on its native 2-degree grid.
- The CPC monthly Nino index on the 1991-2020 baseline (`ersst5_91-20`), Nino 3.4 anomaly column: the main reference.
- The CPC monthly Nino index on the 1981-2010 baseline (`ersst5_81-10`), Nino 3.4 anomaly column: the control.
- The CPC ONI series (`oni`), the official 3-month running index.

Expected numerical results:
- RECON_9120, the monthly Nino 3.4 index reconstructed from ERSSTv5 as the SST anomaly over 5S-5N, 170W-120W relative to a 1991-2020 monthly climatology, 1950-01 to 2023-12. Against the CPC 1991-2020 Nino 3.4 anomaly over that window, report the Pearson correlation to 6 decimal places, the RMSE in degrees Celsius to 4 decimal places, and the largest absolute deviation in degrees Celsius to 4 decimal places with the year and month at which it occurs.
- The number of El Nino events in RECON_9120 over 1950-2023 under the operational ONI criterion (at least 5 consecutive overlapping 3-month seasons, i.e. values of the 3-month centred running mean, at or above +0.5 degrees Celsius), and the three events with the highest peak values, each with the year and month of its peak and its peak value in degrees Celsius to 2 decimal places.
- The same count and the same three largest peaks for the CPC ONI series under the identical criterion, with an explanation of why the two counts differ.
- The mean baseline offset of the reconstruction, RECON_8110 (the same index relative to the 1981-2010 baseline) minus RECON_9120, as the unweighted mean over the months of 1950-01 to 2020-12, in degrees Celsius to 3 decimal places.
- The mean published baseline offset, CPC 1981-2010 minus CPC 1991-2020 Nino 3.4 anomaly, as the unweighted mean over the same months, in degrees Celsius to 3 decimal places, and how closely it agrees with the reconstructed offset.

Expected visualizations:
- Figure 1, a two-row figure sharing a time axis over 1950-2023. The upper row overlays the RECON_9120 and CPC 1991-2020 monthly anomalies, the 3-month running mean, the +0.5 and -0.5 degrees Celsius thresholds, and shading between the threshold and the running-mean curve, warm during detected El Nino events and cool during detected La Nina events (at least 5 consecutive seasons at or below -0.5 degrees Celsius). The lower row shows RECON_9120 minus CPC on its own vertical scale. Label both vertical axes in degrees Celsius.
- Figure 2, a square scatter plot of RECON_9120 (y) against the CPC 1991-2020 series (x), one point per month, with identical axis ranges and ticks, a 1:1 reference line, and the correlation and RMSE annotated.

import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt

DATA = "/data1/lorraine-20419/ocean_data"

# ==================================================================
# 1. ERSSTv5 monthly SST  ->  Nino3.4 region mean (1950-01..2023-12)
# ==================================================================
ds_sst = xr.open_dataset(f"{DATA}/ERSSTv5/sst.mnmean.nc")
sst = ds_sst["sst"]

# native grid: lon 0..358 (0-360 convention), lat descending 88..-88
# Nino3.4 = 5S-5N, 170W-120W  ==  5S-5N, 190E-240E
n34_box = sst.sel(lat=slice(5, -5), lon=slice(190, 240))
n34_da = n34_box.mean(dim=("lat", "lon")).compute()      # simple (unweighted) box mean, as CPC does

tv = n34_da["time"].values
try:
    tidx = pd.DatetimeIndex(tv)
except Exception:                                          # cftime objects
    tidx = pd.DatetimeIndex([pd.Timestamp(int(x.year), int(x.month), 1) for x in tv])

SST_N34 = pd.Series(np.asarray(n34_da.values, dtype=float), index=tidx, name="ersst5_n34")
SST_N34 = SST_N34.loc["1950-01-01":"2023-12-31"].copy()
SST_N34.index = SST_N34.index.to_period("M")
SST_N34 = SST_N34.reindex(pd.period_range("1950-01", "2023-12", freq="M"))

# --- monthly climatologies on the two baselines --------------------
def monthly_clim(s, y0, y1):
    sub = s[(s.index.year >= y0) & (s.index.year <= y1)]
    return sub.groupby(sub.index.month).mean()

clim_9120 = monthly_clim(SST_N34, 1991, 2020)
clim_8110 = monthly_clim(SST_N34, 1981, 2010)

def anomalies(s, clim):
    return pd.Series(s.values - clim.reindex(s.index.month).values, index=s.index)

RECON_9120 = anomalies(SST_N34, clim_9120).rename("RECON_9120")   # main reconstruction
RECON_8110 = anomalies(SST_N34, clim_8110).rename("RECON_8110")   # 1981-2010 control

# ==================================================================
# 2. CPC published Nino indices (1991-2020 and 1981-2010 baselines)
# ==================================================================
def read_nino_monthly(path):
    rows = []
    with open(path) as f:
        for line in f:
            p = line.split()
            if len(p) < 10:
                continue
            try:
                yr, mo = int(p[0]), int(p[1])
                vals = [float(x) for x in p[2:10]]
            except ValueError:                            # header line
                continue
            rows.append([yr, mo] + vals)
    a = np.array(rows, dtype=float)
    a[a <= -90.0] = np.nan                                # -999.9 missing flags
    idx = pd.PeriodIndex(pd.to_datetime(dict(year=a[:, 0].astype(int),
                                             month=a[:, 1].astype(int), day=1)), freq="M")
    cols = ["N12", "N12_ANOM", "N3", "N3_ANOM", "N4", "N4_ANOM", "N34", "N34_ANOM"]
    return pd.DataFrame(a[:, 2:10], index=idx, columns=cols).sort_index()

CPC = read_nino_monthly(f"{DATA}/CPC/ersst5.nino.mth.91-20.ascii")
CPC_9120 = CPC["N34_ANOM"].rename("CPC_9120")
CPC81 = read_nino_monthly(f"{DATA}/CPC/ersst5.nino.mth.81-10.ascii")
CPC_8110 = CPC81["N34_ANOM"].rename("CPC_8110")

# --- CPC ONI (already a 3-month running mean, labelled by season) ---
SEASONS = ["DJF", "JFM", "FMA", "MAM", "AMJ", "MJJ",
           "JJA", "JAS", "ASO", "SON", "OND", "NDJ"]
SEAS_CENTER = {s: i + 1 for i, s in enumerate(SEASONS)}   # DJF -> Jan, JFM -> Feb, ...

rows = []
with open(f"{DATA}/CPC/oni.ascii.txt") as f:
    for line in f:
        p = line.split()
        if len(p) < 4:
            continue
        seas = p[0].upper()
        if seas not in SEAS_CENTER:
            continue
        try:
            yr, anom = int(p[1]), float(p[3])
        except ValueError:
            continue
        rows.append((yr, SEAS_CENTER[seas], anom))

o = pd.DataFrame(rows, columns=["YR", "MON", "ANOM"])
o = o[o["ANOM"] > -90]
ONI = pd.Series(o["ANOM"].values,
                index=pd.PeriodIndex(pd.to_datetime(dict(year=o.YR, month=o.MON, day=1)), freq="M"),
                name="ONI").sort_index()

# ==================================================================
# 3. RECON_9120 vs CPC 1991-2020, 1950-01..2023-12
# ==================================================================
idx = pd.period_range("1950-01", "2023-12", freq="M")
R = RECON_9120.reindex(idx)
C = CPC_9120.reindex(idx)
good = R.notna() & C.notna()
DIFF = (R - C)[good]

corr_recon_cpc = float(np.corrcoef(R[good].values, C[good].values)[0, 1])
rmse_recon_cpc = float(np.sqrt(np.mean(DIFF.values ** 2)))
absdiff = DIFF.abs()
maxdev_time = absdiff.idxmax()                 # Period (year, month) of the largest deviation
maxdev_recon_cpc = float(absdiff.max())

# ==================================================================
# 4. ONI-criterion event detection (>=5 consecutive 3-month seasons >= +0.5)
# ==================================================================
def run3(s):
    return s.rolling(3, center=True).mean()

def detect_events(series, thresh=0.5, min_len=5):
    s = series.dropna().sort_index()
    v = s.values
    n = len(v)
    out = []
    i = 0
    while i < n:
        if v[i] >= thresh:
            j = i
            while j + 1 < n and v[j + 1] >= thresh:
                j += 1
            if (j - i + 1) >= min_len:
                seg = s.iloc[i:j + 1]
                out.append({"start": s.index[i], "end": s.index[j],
                            "peak_time": seg.idxmax(), "peak": float(seg.max()),
                            "n_seasons": int(j - i + 1)})
            i = j + 1
        else:
            i += 1
    return out

RECON3 = run3(RECON_9120)                        # 3-month centred running mean of the reconstruction
EN_RECON = detect_events(RECON3, +0.5, 5)        # El Nino events
LN_RECON = detect_events(-RECON3, +0.5, 5)       # La Nina events (same rule, sign flipped)
en_count_recon = len(EN_RECON)
top3_recon = sorted(EN_RECON, key=lambda e: e["peak"], reverse=True)[:3]

# ONI on the identical window (seasons centred 1950-01..2023-12)
ONI_W = ONI[(ONI.index >= pd.Period("1950-01", "M")) & (ONI.index <= pd.Period("2023-12", "M"))]
EN_ONI = detect_events(ONI_W, +0.5, 5)
LN_ONI = detect_events(-ONI_W, +0.5, 5)
en_count_oni = len(EN_ONI)
top3_oni = sorted(EN_ONI, key=lambda e: e["peak"], reverse=True)[:3]

# ==================================================================
# 5. Baseline offsets, 1950-01..2020-12
# ==================================================================
win = pd.period_range("1950-01", "2020-12", freq="M")
offset_recon = float((RECON_8110.reindex(win) - RECON_9120.reindex(win)).mean())
offset_published = float((CPC_8110.reindex(win) - CPC_9120.reindex(win)).mean())
offset_agreement = offset_recon - offset_published

# ------------------------- reporting ------------------------------
print("RECON_9120 vs CPC 1991-2020 Nino3.4 anomaly (1950-01..2023-12, %d months)" % good.sum())
print("  Pearson r        : %.6f" % corr_recon_cpc)
print("  RMSE             : %.4f degC" % rmse_recon_cpc)
print("  max |deviation|  : %.4f degC at %04d-%02d" % (maxdev_recon_cpc, maxdev_time.year, maxdev_time.month))
print()
print("El Nino events (>=5 consecutive 3-month seasons >= +0.5 degC)")
print("  RECON_9120 : %d" % en_count_recon)
for e in top3_recon:
    print("     peak %04d-%02d  %.2f degC  (event %s..%s, %d seasons)" %
          (e["peak_time"].year, e["peak_time"].month, e["peak"], e["start"], e["end"], e["n_seasons"]))
print("  CPC ONI    : %d" % en_count_oni)
for e in top3_oni:
    print("     peak %04d-%02d  %.2f degC  (event %s..%s, %d seasons)" %
          (e["peak_time"].year, e["peak_time"].month, e["peak"], e["start"], e["end"], e["n_seasons"]))
print("  La Nina events: RECON %d, ONI %d" % (len(LN_RECON), len(LN_ONI)))
print()
print("Mean baseline offset 1981-2010 minus 1991-2020 (1950-01..2020-12)")
print("  reconstructed : %+.3f degC" % offset_recon)
print("  published CPC : %+.3f degC" % offset_published)
print("  agreement     : %+.3f degC" % offset_agreement)

# --- visualization ---
# ================= Figure 1: time series + event shading =================
ts = R.index.to_timestamp()

fig, (ax0, ax1) = plt.subplots(2, 1, figsize=(16, 8.5), sharex=True,
                               gridspec_kw={"height_ratios": [3, 1]})

ax0.plot(ts, R.values, color="0.55", lw=0.7, label="RECON_9120 monthly")
ax0.plot(ts, C.values, color="tab:orange", lw=0.7, alpha=0.85, label="CPC 1991-2020 monthly")

rm = RECON3.dropna()
rm_ts = rm.index.to_timestamp()
ax0.plot(rm_ts, rm.values, color="tab:blue", lw=1.9, label="RECON_9120 3-month running mean")

ax0.axhline(0.0, color="k", lw=0.6)
ax0.axhline(+0.5, color="tab:red", ls="--", lw=1.1)
ax0.axhline(-0.5, color="tab:blue", ls="--", lw=1.1)

warm = pd.Series(False, index=rm.index)
for e in EN_RECON:
    warm.loc[e["start"]:e["end"]] = True
cool = pd.Series(False, index=rm.index)
for e in LN_RECON:
    cool.loc[e["start"]:e["end"]] = True

ax0.fill_between(rm_ts, 0.5, rm.values, where=warm.values, interpolate=True,
                 color="tab:red", alpha=0.30, label="El Nino (5+ seasons >= +0.5)")
ax0.fill_between(rm_ts, -0.5, rm.values, where=cool.values, interpolate=True,
                 color="tab:blue", alpha=0.30, label="La Nina (5+ seasons <= -0.5)")

ax0.set_ylabel("SST anomaly (degC)")
ax0.set_title("Nino 3.4: RECON_9120 vs CPC 1991-2020 and the 3-month running mean, 1950-2023")
ax0.legend(loc="upper left", ncol=3, fontsize=8, frameon=False)

ax1.plot(ts, DIFF.reindex(R.index).values, color="k", lw=0.8)
ax1.axhline(0.0, color="tab:red", ls=":", lw=0.9)
ax1.set_ylabel("RECON - CPC (degC)")
ax1.set_xlabel("Year")

# ================= Figure 2: 1:1 scatter =================
fig2, ax2 = plt.subplots(figsize=(7.5, 7.5))
ax2.scatter(C[good].values, R[good].values, s=12, alpha=0.6,
            color="tab:blue", edgecolors="none")

lo = float(min(R[good].min(), C[good].min())) - 0.25
hi = float(max(R[good].max(), C[good].max())) + 0.25
ax2.plot([lo, hi], [lo, hi], "k--", lw=1.2, label="1:1")
ax2.set_xlim(lo, hi)
ax2.set_ylim(lo, hi)
ax2.set_aspect("equal", adjustable="box")
ticks = np.arange(np.floor(lo * 2.0) / 2.0, hi + 0.25, 0.5)
ax2.set_xticks(ticks)
ax2.set_yticks(ticks)
ax2.grid(alpha=0.3)
ax2.set_xlabel("CPC 1991-2020 Nino3.4 anomaly (degC)")
ax2.set_ylabel("RECON_9120 Nino3.4 anomaly (degC)")
ax2.set_title("Monthly Nino 3.4 anomaly, 1950-2023")
ax2.text(0.03, 0.97,
         "r = %.6f\nRMSE = %.4f degC\nmax|diff| = %.4f degC" %
         (corr_recon_cpc, rmse_recon_cpc, maxdev_recon_cpc),
         transform=ax2.transAxes, va="top", fontsize=10)
ax2.legend(loc="lower right")
