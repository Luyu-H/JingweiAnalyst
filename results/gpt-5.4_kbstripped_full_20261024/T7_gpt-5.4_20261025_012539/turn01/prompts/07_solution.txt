    time_slice="annual",
    depth_range=[200.0, 1000.0],

        depth_range=depth_range,
        chunk_time=12,

    po4_ts = load_regular_data(

        variable="po4",

        depth_range=depth_range,

            dataset="cmip6_historical",
            model=model,
            time_range=time_range,
            depth_range=depth_range,
            variable="thetao",
        )
    else:
        volcello = load_volumes_for_cmip6(
            dataset="cmip6_historical",
            model=model,
            depth_range=depth_range,
            variable="thetao",
        )

    no3_clim = temporal_aggregate(no3_ts, freq="climatology", op="mean").mean("month")
    po4_clim = temporal_aggregate(po4_ts, freq="climatology", op="mean").mean("month")

    no3_umolkg = transform_unit(
        no3_clim,
        from_unit=no3_clim.attrs.get("units", "mol m-3"),
        to_unit="umol kg-1",
        density=1.025,
    )
    po4_umolkg = transform_unit(
        po4_clim,
        from_unit=po4_clim.attrs.get("units", "mol m-3"),
        to_unit="umol kg-1",
        density=1.025,
    )

    sources[model] = {
        "no3_native": no3_clim,
        "po4_native": po4_clim,
        "no3_umolkg": no3_umolkg,
        "po4_umolkg": po4_umolkg,
        "volcello": volcello,
    }

# Compute N:P fields and all requested statistics
for name, src in sources.items():
    no3_native = src["no3_native"]
    po4_native = src["po4_native"]

    np_ratio = compute_ratio(no3_native, po4_native).assign_attrs(units="1")
    src["np_ratio"] = np_ratio

    # Global means over 200-1000 m
    mean_np_headline = compute_basic_stat(
        np_ratio,
        op="volume_weighted_mean",
        dim=None,
    )
    mean_no3 = compute_basic_stat(
        no3_native,
        op="volume_weighted_mean",

    )

    ratio_of_means_val = mean_no3_val / mean_po4_val

    global_stats[name] = {
        "mean_nitrate": mean_no3_val,

        "headline_mean_np": mean_np_headline_val,

    # Regression in umol kg-1 with equal cell weighting

        predictors=[src["po4_umolkg"]],
        method="linear",
        validation="blocked",
    )

    coeffs = reg["coefficients"]
    slope_val = np.nan
    intercept_val = np.nan
    if isinstance(coeffs, dict):
        slope_raw = coeffs.get("predictor_0")
        if slope_raw is None:
            slope_raw = coeffs.get("slope")
        intercept_raw = coeffs.get("intercept")
        if slope_raw is not None:
            slope_val = float(slope_raw)
        if intercept_raw is not None:
            intercept_val = float(intercept_raw)
    else:
        coeffs_arr = np.asarray(coeffs).ravel()
        if coeffs_arr.size >= 1 and coeffs_arr[0] is not None:
            slope_val = float(coeffs_arr[0])
        if coeffs_arr.size >= 2 and coeffs_arr[1] is not None:
            intercept_val = float(coeffs_arr[1])

    regression_stats[name] = {

    # Percent volume with 14 < N:P < 18

            basin_ratio,
            op="volume_weighted_mean",
            dim=None,
        )
        basin_headline_np[name][basin] = float(basin_mean.compute())

    # Percent volume with N:P < 12: global, Pacific, Indian
    below12_stats[name] = {}
    frac_below12_global = compute_exceedance_fraction(

        threshold=12.0,
        comparison="less",
        reference=0.0,
        dim=None,
        weighting="auto",
    )
    below12_stats[name]["global"] = float(np.asarray(frac_below12_global["fraction_percent"]).item())

    for basin in ["Pacific", "Indian"]:
        basin_ratio = select_basin(

            dim=None,
            weighting="auto",
        )

    # 200-1000 m mean N:P minus 16 for map display
    if name == "WOA23":
        mean_np_map = np_ratio.mean("depth")
    else:
        mean_np_map = compute_basic_stat(
            np_ratio,
            op="volume_weighted_mean",
            dim="depth",
        )

Important patterns and insights
- Observed intermediate waters are substantially sub-Redfield on average, but not extremely so: the global mean N:P is about 14.15.
- The observed basin structure matters: Atlantic intermediate waters are much closer to Redfield than Pacific and Indian waters.
- GFDL-ESM4 is the only model that gets close to the observed global mean stoichiometry and global low-N:P volume.
- CESM2 and MPI-ESM1-2-LR both appear too nitrate-depleted relative to phosphate through much of the intermediate ocean.
- Model spread is large, and differences likely reflect both circulation and biogeochemical model structure; the mismatches should not be attributed to transport alone.
