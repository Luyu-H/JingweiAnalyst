The user's question:
Evaluate the 1971-2000 sea surface temperature (SST) climatology of three CMIP6 historical simulations against the World Ocean Atlas 2023 (WOA23) observational climatology, and describe the spatial structure of the bias.

Data to use:
- CMIP6 historical variable `tos` from CESM2, GFDL-ESM4 and MPI-ESM1-2-LR, monthly, from 1971-01 to 2000-12, each on its own native grid with its own published cell areas (areacello).
- WOA23 temperature, the 1971-2000 climate normal, annual time slice, objectively analysed field, at 0 m depth, on the 1-degree grid.
- The WOA23 1-degree land-sea mask (landsea_01), in which a cell is ocean when its value is greater than 1.

Expected numerical results:
- For each model, the global mean of its 1971-2000 SST climatology over its own native ocean grid, in degrees Celsius to 2 decimal places.
- The global mean of the WOA23 0 m annual field over the whole 1-degree grid, in degrees Celsius to 2 decimal places.
- On the WOA23 1-degree grid, over the cells on which the models are compared with WOA23: the mean SST of each model and of WOA23, in degrees Celsius to 2 decimal places.
- On the same cells, for each model: the mean bias delta_SST (model minus WOA23) in degrees Celsius to 2 decimal places, the RMSE against WOA23 in degrees Celsius to 3 decimal places, and the spatial Pearson correlation with WOA23 to 3 decimal places.
- The ranking of the three models from smallest to largest absolute delta_SST.

Expected visualizations:
- Figure 1, a 2-by-2 panel of global maps: the bias fields (model minus WOA23) of CESM2, GFDL-ESM4 and MPI-ESM1-2-LR on the cells of that comparison with one shared diverging colour scale centred on zero, and the WOA23 0 m reference SST on the same cells with a sequential colour scale. Label every panel and give each colour scale a unit label in degrees Celsius.
- Figure 2, a bar chart of the four global mean SST values of the first two items above, each over its own native ocean grid (CESM2, GFDL-ESM4, MPI-ESM1-2-LR and WOA23), each bar annotated with its value, with the y axis labelled in degrees Celsius.

# Computation code only

models = ["CESM2", "GFDL-ESM4", "MPI-ESM1-2-LR"]
time_range = ["1971-01", "2000-12"]

# Containers
tos_monthly = {}
tos_clim_native = {}
native_global_means = {}

# 1) Load CMIP6 monthly tos and form 1971-2000 day-weighted climatological mean on each native grid
for model in models:
    tos_monthly[model] = load_regular_data(
        dataset="cmip6_historical",
        variable="tos",
        model=model,
        time_range=time_range,
    )
    tos_clim_native[model] = temporal_aggregate(
        data=tos_monthly[model],
        freq="mean",
        op="mean",
    )
    native_global_means[model] = float(
        compute_basic_stat(
            data=tos_clim_native[model],
            op="area_weighted_mean",
            dim=["lat", "lon"],
        ).compute()
    )

# 2) Load WOA23 1971-2000 annual temperature climatology at 0 m
woa_surface = load_climatology_data(
    dataset="woa23",
    variable="temperature",
    period="1971-2000",
    time_slice="annual",
    depth_range=[0.0, 0.0],
)

# Drop the singleton depth dimension if present while preserving attrs
if "depth" in woa_surface.dims and woa_surface.sizes["depth"] == 1:
    woa_surface = woa_surface.isel(depth=0, drop=True)

# 3) Load WOA23 land-sea mask and keep ocean where landsea_01 > 1
landsea_01 = load_landsea_mask()
woa_ocean_mask = (landsea_01 > 1)

# Apply ocean mask and valid-data mask to WOA23 field
woa_surface_ocean = woa_surface.where(woa_ocean_mask)

# Native-grid/global mean WOA23 on the 1-degree observational grid
woa_native_global_mean = float(
    compute_basic_stat(
        data=woa_surface_ocean,
        op="area_weighted_mean",
        dim=["lat", "lon"],
    ).compute()
)

# 4) Regrid each model climatology to WOA23 1-degree grid using conservative remapping
tos_regridded = {}
for model in models:
    tos_regridded[model] = regrid_horizontal(
        data=tos_clim_native[model],
        target="woa23",
        method="conservative",
        skipna=True,
        na_thres=0.5,
    )

# 5) Build one shared comparison mask across all three regridded models and WOA23
comparison_mask = build_comparison_mask(
    fields=[tos_regridded[m] for m in models] + [woa_surface_ocean],
    ocean_only=True,
    reduce_other_dims="all",
)

# Apply the common mask
woa_comp = woa_surface_ocean.where(comparison_mask)
tos_comp = {model: tos_regridded[model].where(comparison_mask) for model in models}

# Count comparison cells
comparison_cell_count = int(comparison_mask.sum().compute())

# 6) Shared-grid means and metrics against WOA23
shared_grid_means = {}
difference_metrics = {}
spatial_correlations = {}

for model in models:
    shared_grid_means[model] = {
        "model_mean": float(
            compute_basic_stat(
                data=tos_comp[model],
                op="area_weighted_mean",
                dim=["lat", "lon"],
            ).compute()
        ),
        "woa_mean": float(
            compute_basic_stat(
                data=woa_comp,
                op="area_weighted_mean",
                dim=["lat", "lon"],
            ).compute()
        ),
    }

    difference_metrics[model] = compute_difference_metrics(
        data=tos_comp[model],
        reference=woa_comp,
        dim=["lat", "lon"],
        weighting="area",
    )

    # Materialize scalar outputs
    difference_metrics[model] = {
        k: (float(v.compute()) if hasattr(v, "compute") else v)
        for k, v in difference_metrics[model].items()
        if k != "difference"
    } | {
        "difference": compute_difference_metrics(
            data=tos_comp[model],
            reference=woa_comp,
            dim=["lat", "lon"],
            weighting="area",
        )["difference"]
    }

    spatial_correlations[model] = analyze_correlation(
        var1=tos_comp[model],
        var2=woa_comp,
        dim=["lat", "lon"],
        method="pearson",
        weighting="area",
    )

    spatial_correlations[model] = {
        k: (float(v.compute()) if hasattr(v, "compute") else v)
        for k, v in spatial_correlations[model].items()
    }

# 7) Ranking by absolute mean bias
abs_bias_ranking = sorted(
    models,
    key=lambda m: abs(difference_metrics[m]["bias"])
)

# 8) Compact report table with requested precision
summary_rows = []
for model in models:
    summary_rows.append({
        "model": model,
        "native_global_mean_SST_degC": round(native_global_means[model], 2),
        "shared_grid_model_mean_degC": round(shared_grid_means[model]["model_mean"], 2),
        "shared_grid_WOA23_mean_degC": round(shared_grid_means[model]["woa_mean"], 2),
        "delta_SST_degC": round(difference_metrics[model]["bias"], 2),
        "RMSE_degC": round(difference_metrics[model]["rmse"], 3),
        "spatial_r_degCgrid": round(spatial_correlations[model]["correlation"], 3),
        "n_comp_cells": int(difference_metrics[model]["n_valid"]),
    })

results_table = pd.DataFrame(summary_rows)
woa_native_global_mean_rounded = round(woa_native_global_mean, 2)
abs_bias_ranking_with_values = [
    (m, round(difference_metrics[m]["bias"], 2), round(abs(difference_metrics[m]["bias"]), 2))
    for m in abs_bias_ranking
]

# 9) Bias fields for plotting
bias_fields = {}
for model in models:
    bias_fields[model] = (tos_comp[model] - woa_comp).assign_attrs(units="degC")

# 10) Variables explicitly available for downstream reporting / plotting
cesm2_native_global_mean = native_global_means["CESM2"]
gfdl_native_global_mean = native_global_means["GFDL-ESM4"]
mpi_native_global_mean = native_global_means["MPI-ESM1-2-LR"]

cesm2_bias = difference_metrics["CESM2"]["bias"]
gfdl_bias = difference_metrics["GFDL-ESM4"]["bias"]
mpi_bias = difference_metrics["MPI-ESM1-2-LR"]["bias"]

cesm2_rmse = difference_metrics["CESM2"]["rmse"]
gfdl_rmse = difference_metrics["GFDL-ESM4"]["rmse"]
mpi_rmse = difference_metrics["MPI-ESM1-2-LR"]["rmse"]

cesm2_corr = spatial_correlations["CESM2"]["correlation"]
gfdl_corr = spatial_correlations["GFDL-ESM4"]["correlation"]
mpi_corr = spatial_correlations["MPI-ESM1-2-LR"]["correlation"]

# --- visualization ---
# Visualization code only

# --- Figure 1: 2x2 panel of bias maps and WOA23 SST map ---
fig1, axes = plt.subplots(2, 2, figsize=(15, 8), constrained_layout=True)

# Robust shared color scale for bias panels
all_bias_vals = np.concatenate([
    bias_fields["CESM2"].values.ravel(),
    bias_fields["GFDL-ESM4"].values.ravel(),
    bias_fields["MPI-ESM1-2-LR"].values.ravel(),
])
all_bias_vals = all_bias_vals[np.isfinite(all_bias_vals)]
bias_vmax = float(np.nanpercentile(np.abs(all_bias_vals), 98))

# Robust scale for WOA23 SST
woa_vals = woa_comp.values.ravel()
woa_vals = woa_vals[np.isfinite(woa_vals)]
woa_vmin = float(np.nanpercentile(woa_vals, 2))
woa_vmax = float(np.nanpercentile(woa_vals, 98))

panel_order = [
    ("CESM2", axes[0, 0]),
    ("GFDL-ESM4", axes[0, 1]),
    ("MPI-ESM1-2-LR", axes[1, 0]),
]

bias_mappables = []
for model, ax in panel_order:
    im = ax.pcolormesh(
        bias_fields[model]["lon"],
        bias_fields[model]["lat"],
        bias_fields[model],
        cmap="RdBu_r",
        shading="auto",
        vmin=-bias_vmax,
        vmax=bias_vmax,
    )
    bias_mappables.append(im)
    ax.set_title(f"{model} - WOA23 SST bias")
    ax.set_xlabel("Longitude")
    ax.set_ylabel("Latitude")
    ax.set_xlim(float(bias_fields[model]["lon"].min()), float(bias_fields[model]["lon"].max()))
    ax.set_ylim(float(bias_fields[model]["lat"].min()), float(bias_fields[model]["lat"].max()))

im_woa = axes[1, 1].pcolormesh(
    woa_comp["lon"],
    woa_comp["lat"],
    woa_comp,
    cmap="viridis",
    shading="auto",
    vmin=woa_vmin,
    vmax=woa_vmax,
)
axes[1, 1].set_title("WOA23 0 m annual SST")
axes[1, 1].set_xlabel("Longitude")
axes[1, 1].set_ylabel("Latitude")
axes[1, 1].set_xlim(float(woa_comp["lon"].min()), float(woa_comp["lon"].max()))
axes[1, 1].set_ylim(float(woa_comp["lat"].min()), float(woa_comp["lat"].max()))

cbar_bias = fig1.colorbar(
    bias_mappables[0],
    ax=[axes[0, 0], axes[0, 1], axes[1, 0]],
    orientation="vertical",
    shrink=0.9,
    extend="both",
)
cbar_bias.set_label("SST bias (°C)")

cbar_woa = fig1.colorbar(
    im_woa,
    ax=axes[1, 1],
    orientation="vertical",
    shrink=0.9,
    extend="both",
)
cbar_woa.set_label("SST (°C)")

fig1.suptitle("Figure 1. 1971-2000 SST climatology biases on shared WOA23 1° comparison cells (98th percentile clipping)")

# --- Figure 2: bar chart of own-grid global means ---
fig2, ax = plt.subplots(figsize=(8, 5), constrained_layout=True)

bar_labels = ["CESM2", "GFDL-ESM4", "MPI-ESM1-2-LR", "WOA23"]
bar_values = [
    native_global_means["CESM2"],
    native_global_means["GFDL-ESM4"],
    native_global_means["MPI-ESM1-2-LR"],
    woa_native_global_mean,
]

bars = ax.bar(bar_labels, bar_values, color=["tab:blue", "tab:orange", "tab:green", "tab:gray"])
ax.set_ylabel("Global mean SST (°C)")
ax.set_title("Figure 2. Global mean 1971-2000 SST climatology on each native ocean grid")

for bar, val in zip(bars, bar_values):
    ax.text(

    )
