def depth_name(da):
    for c in ("depth", "lev", "olevel", "zlev"):
        if c in da.dims:
            return c
    raise ValueError("no depth dimension found in " + str(da.dims))


def vol_weighted_sss(field, weight):
    """Within-cell mean over the retained levels, each level weighted by its FULL volume."""
    d = depth_name(field)
    valid = field.notnull()
    w = xr.where(valid, weight, 0.0).fillna(0.0)
    num = (field.fillna(0.0) * w).sum(dim=d)
    den = w.sum(dim=d)
    out = num / den
    attrs = dict(field.attrs)
    attrs.update(units="psu",
                 long_name="sea surface salinity (levels shallower than 10 m, volume weighted)")
    return out.assign_attrs(**attrs)

def depth_name(da):
    for c in ("depth", "lev", "olevel", "zlev"):
        if c in da.dims:
            return c
    raise ValueError("no depth dimension found in " + str(da.dims))


def vol_weighted_sss(field, weight):
    """Within-cell mean over the retained levels, each level weighted by its FULL volume."""
    d = depth_name(field)
    w = weight

    # The weight may be published per month (a time-varying volcello): a weight cannot
    # be aligned against the data step by step, so collapse it to a static field first.
    if "time" in w.dims:
        w = w.mean(dim="time", skipna=True)
    if "time" in w.coords:
        w = w.drop_vars("time")

    # ... and it may name its vertical axis differently from the field.
    for c in ("depth", "lev", "olevel", "zlev"):
        if c in w.dims and c != d:
            w = w.rename({c: d})
            break

    # Put the weight on exactly the levels of the field, so that the two align exactly
    # (a float32/float64 difference in the level labels would otherwise be fatal).
    if d in w.dims:
        w = w.sel({d: field[d].values}, method="nearest")

    valid = field.notnull()
    w = xr.where(valid, w, 0.0).fillna(0.0)
    num = (field.fillna(0.0) * w).sum(dim=d)
    den = w.sum(dim=d)
    out = num / den
    attrs = dict(field.attrs)
    attrs.update(units="psu",
                 long_name="sea surface salinity (levels shallower than 10 m, volume weighted)")
    return out.assign_attrs(**attrs)
