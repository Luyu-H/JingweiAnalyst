def _scalar(x):
    """Materialise a scalar-like xarray/dask/numpy object."""
    if hasattr(x, "compute"):
        x = x.compute()
    size = int(getattr(x, "size", 1))
    if size == 1 and hasattr(x, "item"):
        return float(x.item())
    if size == 1:
        return float(np.asarray(x).reshape(-1)[0])
    return float(x)
